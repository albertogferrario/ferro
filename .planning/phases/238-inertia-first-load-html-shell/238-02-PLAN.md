```rust
#[cfg(test)]
mod content_negotiation_tests {
    use super::*;
    use crate::config::InertiaConfig;

    struct MockReq { is_inertia: bool, path: &'static str }
    // impl crate::request::InertiaRequest for MockReq { ... }  // match the real trait

    #[test]
    fn non_inertia_request_returns_html_document() {
        let req = MockReq { is_inertia: false, path: "/home" };
        let resp = Inertia::render_with_config(&req, "Home",
            serde_json::json!({"title": "Hi"}), InertiaConfig::new().development());
        assert_eq!(resp.content_type, "text/html; charset=utf-8");
        assert!(resp.body.contains("<!DOCTYPE html>"));
        assert!(resp.body.contains(r#"data-page=""#));
    }

    #[test]
    fn inertia_request_returns_json_contract() {
        let req = MockReq { is_inertia: true, path: "/home" };
        let resp = Inertia::render_with_config(&req, "Home",
            serde_json::json!({"title": "Hi"}), InertiaConfig::new().development());
        assert_eq!(resp.content_type, "application/json");
        let body: serde_json::Value = serde_json::from_str(&resp.body).unwrap();
        assert_eq!(body["component"], "Home");
    }

    #[test]
    fn html_data_page_equals_json_contract() {
        let props = serde_json::json!({"title": "Hi", "count": 42});
        let cfg = InertiaConfig::new().development().version("test-1");
        let html = Inertia::render_with_config(
            &MockReq { is_inertia: false, path: "/home" }, "Home", props.clone(), cfg.clone());
        let json = Inertia::render_with_config(
            &MockReq { is_inertia: true, path: "/home" }, "Home", props, cfg);
        let start = html.body.find(r#"data-page=""#).unwrap() + 11;
        let end = html.body[start..].find('"').unwrap() + start;
        let page = html.body[start..end]
            .replace("&quot;", "\"").replace("&amp;", "&")
            .replace("&lt;", "<").replace("&gt;", ">").replace("&#x27;", "'");
        let html_page: serde_json::Value = serde_json::from_str(&page).unwrap();
        let json_page: serde_json::Value = serde_json::from_str(&json.body).unwrap();
        assert_eq!(html_page, json_page);
    }

    #[test]
    fn dev_mode_emits_vite_client_script() {
        let resp = Inertia::render_with_config(
            &MockReq { is_inertia: false, path: "/" }, "App", serde_json::json!({}),
            InertiaConfig::new().development().vite_dev_server("http://localhost:5173"));
        assert!(resp.body.contains("http://localhost:5173/@vite/client"));
    }

    #[test]
    fn title_override() {
        let resp = Inertia::render_with_config(
            &MockReq { is_inertia: false, path: "/" }, "App", serde_json::json!({}),
            InertiaConfig::new().development().app_name("Fallback").title("Explicit"));
        assert!(resp.body.contains("<title>Explicit</title>"));
        assert!(!resp.body.contains("<title>Fallback</title>"));
    }

    #[test]
    fn head_extras_in_html() {
        let resp = Inertia::render_with_config(
            &MockReq { is_inertia: false, path: "/" }, "App", serde_json::json!({}),
            InertiaConfig::new().development().head_extras(r#"<meta name="x" content="y">"#));
        assert!(resp.body.contains(r#"<meta name="x" content="y">"#));
    }

    #[test]
    fn mount_id_applied() {
        let resp = Inertia::render_with_config(
            &MockReq { is_inertia: false, path: "/" }, "App", serde_json::json!({}),
            InertiaConfig::new().development().mount_id("root"));
        assert!(resp.body.contains(r#"id="root" data-page="#));
    }

    #[test]
    fn prod_mode_does_not_leak_dev_server() {
        // SECURITY (T-238-03): prod build must never emit the @vite/client preamble.
        let resp = Inertia::render_with_config(
            &MockReq { is_inertia: false, path: "/" }, "App", serde_json::json!({}),
            InertiaConfig::new().production().vite_dev_server("http://localhost:5173"));
        assert!(!resp.body.contains("/@vite/client"));
        assert!(!resp.body.contains("@react-refresh"));
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| props → data-page HTML attribute | serialized props embedded into a double-quoted HTML attribute |
| config.head_extras → `<head>` | raw developer-controlled HTML injected verbatim |
| dev-mode tags → HTML | Vite dev-server URL + react-refresh preamble must only emit in dev |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-238-01 | Tampering (XSS) | `data-page` attribute embedding props | mitigate | The existing escaping (response.rs:382-389) encodes `& < > " '` — covering the double-quoted attribute context. This plan does NOT modify that block; `html_data_page_equals_json_contract` round-trips through unescaping to prove no regression. |
| T-238-02 | Tampering (XSS) | `head_extras` raw HTML in `<head>` | accept (with documented boundary) | head_extras is developer-controlled config, NOT request input. Field doc (Plan 01) forbids populating it from request data. No request-data path reaches it. Injection is verbatim by design (it carries meta/favicon/font tags). |
| T-238-03 | Information disclosure | dev-server URL / `@react-refresh` preamble leaking into prod | mitigate | Dev tags are gated behind `self.config.development`. `prod_mode_does_not_leak_dev_server` test asserts the prod branch emits neither `/@vite/client` nor `@react-refresh`. |
</threat_model>
