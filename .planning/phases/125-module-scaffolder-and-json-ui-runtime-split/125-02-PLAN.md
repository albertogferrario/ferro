<objective>
Refactor `ferro-json-ui/src/runtime.rs` from a 600-line monolithic IIFE into
per-concern Rust submodules under `ferro-json-ui/src/runtime/`, each holding a
JS source string constant for one named setup function. A `mod.rs` assembler
concatenates them into the same single emitted bundle (zero extra HTTP
requests), wrapped by a `ferroRuntime()` dispatcher invoked on DOMContentLoaded.
Honors D-06..D-11.

Purpose: Make the runtime testable and editable per-feature without touching
unrelated code. Public API (`crate::runtime::FERRO_RUNTIME_JS`) is unchanged.
Output: New `runtime/` directory tree, refactored emitted bundle, expanded
unit test suite, identical UI behavior.
</objective>

       //! Built-in JavaScript runtime for ferro-json-ui (split per concern).
       //!
       //! Each submodule contributes one `setup*` function. The assembled
       //! bundle wraps them in a single IIFE with a `ferroRuntime()`
       //! dispatcher invoked on DOMContentLoaded. The emitted output is a
       //! single string — no extra HTTP requests.

       mod sse;
       mod toasts;
       mod dismissibles;
       mod notifications;
       mod tabs;
       mod sidebar;
       mod form_guards;
       mod dropdowns;
       mod product_tiles;
       mod modals;
       mod kanban;

       pub(crate) const FERRO_RUNTIME_JS: &str = concat!(
           "(function() {\n    'use strict';\n",
           sse::SOURCE,
           tabs::SOURCE,
           toasts::SOURCE,
           dismissibles::SOURCE,
           notifications::SOURCE,
           dropdowns::SOURCE,
           modals::SOURCE,
           sidebar::SOURCE,
           form_guards::SOURCE,
           product_tiles::SOURCE,
           kanban::SOURCE,
           "\n    function ferroRuntime() {\n",
           "        setupSSE();\n",
           "        setupTabs();\n",
           "        setupDismissibles();\n",
           "        setupNotifications();\n",
           "        setupDropdowns();\n",
           "        setupKanban();\n",
           "        setupSidebar();\n",
           "        setupFormGuards();\n",
           "        setupProductTiles();\n",
           "        setupModals();\n",
           "        setupToasts();\n",
           "    }\n",
           "    document.addEventListener('DOMContentLoaded', ferroRuntime);\n",
           "})();\n",
       );
       ```

       fn bundle_contains_dispatcher() {
           assert!(FERRO_RUNTIME_JS.contains("function ferroRuntime()"));
           assert!(FERRO_RUNTIME_JS.contains("DOMContentLoaded"));
           assert!(FERRO_RUNTIME_JS.contains("ferroRuntime"));

           for fn_name in [
               "setupSSE", "setupTabs", "setupToasts", "setupSidebar",
               "setupDropdowns", "setupModals", "setupDismissibles",
               "setupNotifications", "setupFormGuards", "setupProductTiles",
               "setupKanban",
           ] {

                   FERRO_RUNTIME_JS.contains(fn_name),
                   "bundle missing {}", fn_name
               );
           }
       }

       #[test]
       fn bundle_is_single_iife() {
           // exactly one opening IIFE wrapper
           assert_eq!(FERRO_RUNTIME_JS.matches("(function() {").count(), 1);
           assert!(FERRO_RUNTIME_JS.trim_end().ends_with("})();"));
       }

       #[test]
       fn dispatcher_invokes_every_setup() {
           // The dispatcher body must call every setup function
           let dispatcher_start = FERRO_RUNTIME_JS.find("function ferroRuntime()").unwrap();
           let dispatcher = &FERRO_RUNTIME_JS[dispatcher_start..];

               "setupDropdowns();", "setupModals();", "setupDismissibles();",
               "setupNotifications();", "setupFormGuards();", "setupProductTiles();",
               "setupKanban();",
           ] {
               assert!(dispatcher.contains(call), "dispatcher missing {}", call);
           }
       }
       ```
