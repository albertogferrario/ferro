         ```rust
         fn render_envs_block(env_example_contents: &str) -> String {
             let lines = parse_env_example_structured(env_example_contents);
             let mut out = String::new();
             let indent = "  "; // child of `envs:` — adjust to match template indentation
             for line in lines {
                 match line {
                     EnvLine::Key(key) => {
                         out.push_str(indent);
                         out.push_str("- key: ");
                         out.push_str(&key);
                         out.push('\n');
                         out.push_str(indent);
                         out.push_str("  value: \"\"\n");
                         if is_secret_key(&key) {
                             out.push_str(indent);
                             out.push_str("  type: SECRET\n");
                             out.push_str(indent);
                             out.push_str("  scope: RUN_AND_BUILD_TIME\n");
                         } else {
                             out.push_str(indent);
                             out.push_str("  scope: RUN_TIME\n");
                         }
                     }
                     EnvLine::Blank => {
                         out.push('\n');
                     }
                     EnvLine::Comment => {
                         // comments in .env.example are dropped from the YAML
                         // block; blank-line separators carry the grouping.
                     }
                 }
             }
             // Strip trailing newline to fit cleanly into the template slot.
             while out.ends_with('\n') {
                 out.pop();
             }
             out
         }
         ```

       ```rust
       mod envs_block_tests {
           use super::*;

           #[test]
           fn envs_block_from_env_example() {
               let src = "DATABASE_URL=\nSTRIPE_SECRET_KEY=\nAPP_NAME=\n";
               let out = render_envs_block(src);
               assert!(out.contains("- key: DATABASE_URL"));
               assert!(out.contains("- key: STRIPE_SECRET_KEY"));
               assert!(out.contains("- key: APP_NAME"));
           }

           #[test]
           fn secret_scope_and_type() {
               let src = "DATABASE_URL=\nSTRIPE_SECRET_KEY=\nAPP_NAME=\n";
               let out = render_envs_block(src);
               // Secret: has type: SECRET AND RUN_AND_BUILD_TIME
               let stripe_idx = out.find("- key: STRIPE_SECRET_KEY").unwrap();
               let stripe_block = &out[stripe_idx..];
               let stripe_block_end = stripe_block
                   .find("- key: ")
                   .map(|_| stripe_block.find("\n\n").unwrap_or(stripe_block.len()))
                   .unwrap_or(stripe_block.len());
               let stripe_slice = &stripe_block[..stripe_block_end.min(stripe_block.len())];
               assert!(stripe_slice.contains("type: SECRET"));
               assert!(stripe_slice.contains("scope: RUN_AND_BUILD_TIME"));

               // Non-secret: no type:, scope RUN_TIME
               let db_idx = out.find("- key: DATABASE_URL").unwrap();
               let db_end = out[db_idx..].find("- key: ").map(|e| db_idx + e).unwrap_or(out.len());
               let db_slice = &out[db_idx..db_end];
               assert!(!db_slice.contains("type: SECRET"));
               assert!(db_slice.contains("scope: RUN_TIME"));
           }

           #[test]
           fn envs_preserve_source_order() {
               let src = "Z_KEY=\nA_KEY=\nM_KEY=\n";
               let out = render_envs_block(src);
               let z = out.find("Z_KEY").unwrap();
               let a = out.find("A_KEY").unwrap();
               let m = out.find("M_KEY").unwrap();
               assert!(z < a && a < m);
           }

           #[test]
           fn envs_preserve_blank_separators() {
               let src = "A=\n\nB=\n";
               let out = render_envs_block(src);
               // A block, blank line, B block
               let a = out.find("- key: A").unwrap();
               let b = out.find("- key: B").unwrap();
               assert!(out[a..b].contains("\n\n"));
           }
       }
       ```
