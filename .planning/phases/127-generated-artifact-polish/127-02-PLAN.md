           fn dockerignore_whitelists_readme() {

               let idx_whitelist = out.find("!README.md").unwrap();
               let before = &out[..idx_whitelist];
               let last_line = before.lines().rev().find(|l| !l.trim().is_empty()).unwrap_or("");
               assert!(last_line.trim_start().starts_with('#'), "expected comment above !README.md, got: {}", last_line);
