Implement the `#[action]` proc-macro in `ferro-macros/src/action.rs`. The macro transforms an async function returning `ActionResult` into a `framework::Response`-returning function by extracting parameters (using the Plan 02 `crate::utils` helpers) and dispatching to the Plan 01 runtime helper `handle_action_result`.

Purpose: This is the user-facing entry point of the killer feature. Once landed, a user writes:

```rust
#[action(redirect_to = "/dashboard/pagine")]
pub async fn publish_by_id(req: Request) -> ActionResult {
    let business = resolve_tenant().await?;
    let id: i64 = req.param("id")?.parse()?;
    let frontend = Page::find_by_id(id).await?
        .ok_or(ActionError::msg("Pagina non trovata"))?;
    if frontend.tenant_id != business.id { return Err(ActionError::forbidden("")); }
    publish_page(...).await?;
    Page::save_validation_json(...).await?;
    Page::mark_published(...).await?;
    Ok(())
}
```

       Full file contents:

       ```rust
       //! `#[action]` attribute macro implementation.
       //!
       //! Transforms an async handler that returns `ActionResult` into a
       //! `ferro::Response`-returning handler. The macro:
       //!
       //! - Parses required `redirect_to = "..."` and optional `method = "..."`.
       //! - Extracts parameters using the shared scaffold in `crate::utils`
       //!   (same as `#[handler]`).
       //! - Wraps the user body in an `ActionResult`-typed scope so `?` works.
       //! - Dispatches the result to `ferro::http::action::handle_action_result`,
       //!   which emits the 303 redirect, writes session flash, and logs.
       //!
       //! Generated handler shape:
       //!
       //! ```ignore
       //! pub async fn h(__ferro_req: ::ferro::Request) -> ::ferro::Response {
       //!     let mut __ferro_req = __ferro_req;
       //!     let __ferro_params = __ferro_req.params().clone();
       //!     /* extractions for user-declared params */
       //!     let __action_result: ::ferro::ActionResult = { /* user body */ };
       //!     ::ferro::http::action::handle_action_result(
       //!         __action_result,
       //!         "<redirect_to literal>",
       //!         concat!(module_path!(), "::", stringify!(h)),
       //!         &mut __ferro_req,
       //!     )
       //! }
       //! ```

       use proc_macro::TokenStream;
       use proc_macro2::Span;
       use quote::quote;
       use syn::{parse_macro_input, Expr, FnArg, ItemFn, Lit, Meta};

       use crate::utils::{classify_param_type, extract_param_name, ferro, generate_extraction};

       /// Parsed attributes from `#[action(redirect_to = "...", method = "POST")]`.
       struct ActionAttrs {
           redirect_to: String,
           /// HTTP method default `"POST"`. Currently unused at codegen time (the
           /// macro emits an unconditional handler), but parsed so the surface
           /// matches CONTEXT D-05 and future routing integration can read it.
           #[allow(dead_code)]
           method: String,
       }

       fn parse_action_attrs(attr: TokenStream) -> Result<ActionAttrs, syn::Error> {
           let mut redirect_to: Option<String> = None;
           let mut method: String = "POST".to_string();

           let parser = syn::punctuated::Punctuated::<Meta, syn::Token![,]>::parse_terminated;
           let metas = syn::parse::Parser::parse(parser, attr)
               .map_err(|e| syn::Error::new(e.span(), format!("#[action]: invalid attribute syntax: {e}")))?;

           for meta in metas {
               match meta {
                   Meta::NameValue(nv) => {
                       let key = nv.path.get_ident().map(|i| i.to_string()).unwrap_or_default();
                       match key.as_str() {
                           "redirect_to" => {
                               if let Expr::Lit(expr_lit) = &nv.value {
                                   if let Lit::Str(lit_str) = &expr_lit.lit {
                                       redirect_to = Some(lit_str.value());
                                       continue;
                                   }
                               }
                               return Err(syn::Error::new_spanned(
                                   &nv.value,
                                   "#[action]: `redirect_to` must be a string literal",
                               ));
                           }
                           "method" => {
                               if let Expr::Lit(expr_lit) = &nv.value {
                                   if let Lit::Str(lit_str) = &expr_lit.lit {
                                       method = lit_str.value();
                                       continue;
                                   }
                               }
                               return Err(syn::Error::new_spanned(
                                   &nv.value,
                                   "#[action]: `method` must be a string literal (default \"POST\")",
                               ));
                           }
                           other => {
                               return Err(syn::Error::new_spanned(
                                   nv.path,
                                   format!(
                                       "#[action]: unknown attribute `{other}` — supported keys: `redirect_to`, `method`"
                                   ),
                               ));
                           }
                       }
                   }
                   other => {
                       return Err(syn::Error::new_spanned(
                           other,
                           "#[action]: only `key = \"value\"` attributes are supported",
                       ));
                   }
               }
           }

           let redirect_to = redirect_to.ok_or_else(|| {
               syn::Error::new(
                   Span::call_site(),
                   "#[action]: `redirect_to` is required, e.g. #[action(redirect_to = \"/dashboard/foo\")]",
               )
           })?;

           Ok(ActionAttrs { redirect_to, method })
       }

       /// Implementation of the `#[action]` attribute macro.
       pub fn action_impl(attr: TokenStream, input: TokenStream) -> TokenStream {
           let attrs = match parse_action_attrs(attr) {
               Ok(a) => a,
               Err(e) => return e.to_compile_error().into(),
           };

           let input_fn = parse_macro_input!(input as ItemFn);

           let ferro = ferro();

           let fn_vis = &input_fn.vis;
           let fn_name = &input_fn.sig.ident;
           let fn_generics = &input_fn.sig.generics;
           let fn_block = &input_fn.block;
           let fn_attrs = &input_fn.attrs;

           // Generate parameter extractions using the shared scaffold.
           let params: Vec<_> = input_fn.sig.inputs.iter().collect();
           let mut extractions = Vec::new();
           let mut has_request_consumer = false;
           let mut has_request_param = false;

           for param in &params {
               match param {
                   FnArg::Typed(pat_type) => {
                       let param_pat = &pat_type.pat;
                       let param_type = &pat_type.ty;
                       let param_name = extract_param_name(param_pat);
                       let kind = classify_param_type(param_type);
                       let extraction = generate_extraction(
                           &ferro,
                           param_pat,
                           param_type,
                           &param_name,
                           &kind,
                           &mut has_request_consumer,
                           &mut has_request_param,
                       );
                       extractions.push(extraction);
                   }
                   FnArg::Receiver(_) => {
                       return syn::Error::new_spanned(
                           param,
                           "#[action] does not support methods with `self` receiver",
                       )
                       .to_compile_error()
                       .into();
                   }
               }
           }

           // Discourage `has_request_consumer` unused warning when no params declared.
           let _ = has_request_consumer;
           let _ = has_request_param;

           let redirect_to_lit = &attrs.redirect_to;

           // Generated function:
           //   pub async fn #fn_name<...>(__ferro_req: ::ferro::Request) -> ::ferro::Response {
           //       let mut __ferro_req = __ferro_req;
           //       let __ferro_params = __ferro_req.params().clone();
           //       /* extractions */
           //       let __action_result: #ferro::ActionResult = { #fn_block };
           //       #ferro::http::action::handle_action_result(
           //           __action_result,
           //           #redirect_to_lit,
           //           concat!(module_path!(), "::", stringify!(#fn_name)),
           //           &mut __ferro_req,
           //       )
           //   }
           //
           // The `let mut __ferro_req = __ferro_req;` shadowing line is intentional:
           // - It makes `req.flash(...)` / `req.redirect_to(...)` calls inside the
           //   user body mutate the request's override state (the user-declared
           //   `req` binding aliases `__ferro_req` for the `Request` param case via
           //   `generate_extraction` in `crate::utils`).
           // - For the `Request` param case, `generate_extraction` emits
           //   `let #pat: #ty = __ferro_req;` which moves the request — so the user
           //   binding `req` ends up owning the request and `req.flash(...)` works
           //   on `req` directly. But we still need `&mut __ferro_req` for the
           //   final `handle_action_result` call. The shadowing above is the
           //   simplest correct construction; if `generate_extraction` moves the
           //   request into the user's `req` binding, we need to thread the request
           //   back to the runtime helper. Resolution: see Action note below.

           let output = quote! {
               #(#fn_attrs)*
               #fn_vis async fn #fn_name #fn_generics(__ferro_req: #ferro::Request) -> #ferro::Response {
                   let mut __ferro_req = __ferro_req;
                   let __ferro_params = __ferro_req.params().clone();
                   #(#extractions)*
                   let __action_result: #ferro::ActionResult = { #fn_block };
                   #ferro::http::action::handle_action_result(
                       __action_result,
                       #redirect_to_lit,
                       concat!(module_path!(), "::", stringify!(#fn_name)),
                       &mut __ferro_req,
                   )
               }
           };

           output.into()
       }
       ```

       **Critical resolution before emitting code — Request binding ownership:**

       The `#[handler]` `generate_extraction` for `ParamKind::Request` emits `let #pat: #ty = __ferro_req;` which MOVES `__ferro_req` into the user-declared binding (e.g., `req`). After that statement, `__ferro_req` is consumed and cannot be passed to `handle_action_result` as `&mut __ferro_req`.

       Two options, pick the simpler:

       **Option A — Re-bind by reference (RECOMMENDED).** For `#[action]`, do NOT reuse `generate_extraction` directly for the `Request` param case. Instead, before the `#(#extractions)*` block, generate:

       ```rust
       // Action-specific: bind user's `req` as &mut without moving __ferro_req out of scope.
       let #user_req_pat: &mut #ferro::Request = &mut __ferro_req;
       ```

       Then SKIP the `Request` extraction in the loop (i.e., do not call `generate_extraction` for `ParamKind::Request` — emit nothing for it, since the rebinding above covers it). For non-Request params, `generate_extraction` works unchanged (they extract from `__ferro_params` or from `__ferro_req` by reference internally via `FromRequest`).

       **Critical: the user-declared `req` binding does NOT need to be `mut`** — `Request::flash(...)` and `Request::redirect_to(...)` are `&mut self`, which is satisfied because `req: &mut Request` (the binding type is `&mut #ferro::Request`). The `&mut` borrow gives the user method-call mutability without making the binding itself `mut`. If the user writes `let req: &mut Request = ...` and then calls `req.flash("x")`, that compiles because method-call autoref takes `&mut *req`.

       However: the `&mut __ferro_req` borrow MUST live for the duration of the user body AND be released before the final `handle_action_result` call also takes `&mut __ferro_req`.
