       ```rust
       //! Integration tests for the `#[action]` runtime helper.
       //!
       //! Exercises `__test_handle_action_result` (the doc-hidden test handle for
       //! `handle_action_result`) against simulated `Ok(())` and `Err(ActionError::...)`
       //! inputs, asserting on:
       //!
       //! - 303 Location header (happy path)
       //! - Success-side overrides via `req.flash(...)` / `req.redirect_to(...)` (D-02)
       //! - Error-side `redirect_override` via `ActionError::*::redirect_to(...)` (D-01)
       //! - T-180-02 open-redirect mitigation (both success and error paths)
       //! - T-180-03 log-injection mitigation (sanitizer strips control chars)
       //! - Back-compat query string (D-06)
       //! - Session flash payload shape `{variant, message}` (D-06)

       extern crate ferro_rs as ferro;

       use ferro::{ActionError, ActionResult, FlashVariant, Request};

       /// Read the Location header value from the HttpResponse via the verified
       /// getter `HttpResponse::headers() -> &[(String, String)]` (response.rs:142).
       /// Returns `None` if no Location header is present. Case-insensitive on the
       /// header name to mirror RFC 7230 § 3.2.
       fn location_header(resp: &ferro::HttpResponse) -> Option<&str> {
           resp.headers()
               .iter()
               .find(|(k, _)| k.eq_ignore_ascii_case("location"))
               .map(|(_, v)| v.as_str())
       }

       /// Unwrap a `ferro::Response` (which is `Result<HttpResponse, HttpResponse>`)
       /// for inspection.
       fn unwrap_response(resp: &ferro::Response) -> &ferro::HttpResponse {
           match resp {
               Ok(r) => r,
               Err(r) => r,
           }
       }

       /// Construct a real `ferro::Request` via TCP loopback — the canonical pattern
       /// from `framework/src/tenant/mod.rs:160-200`. `Request::new` requires a
       /// `hyper::Request<hyper::body::Incoming>` and `Incoming` has no default
       /// constructor, so we accept an actual hyper connection. The executor MAY
       /// reuse a public `framework::testing::make_request()` helper if one exists.
       async fn make_request() -> Request {
           use hyper_util::rt::TokioIo;
           use tokio::net::TcpListener;
           use tokio::sync::oneshot;

           let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
           let addr = listener.local_addr().unwrap();

           let (tx, rx) = oneshot::channel::<Request>();
           let tx_holder = std::sync::Arc::new(std::sync::Mutex::new(Some(tx)));

           tokio::spawn(async move {
               if let Ok((stream, _)) = listener.accept().await {
                   let io = TokioIo::new(stream);
                   let tx_holder = tx_holder.clone();
                   hyper::server::conn::http1::Builder::new()
                       .serve_connection(
                           io,
                           hyper::service::service_fn(move |req| {
                               let tx_holder = tx_holder.clone();
                               async move {
                                   if let Some(tx) = tx_holder.lock().unwrap().take() {
                                       let _ = tx.send(Request::new(req));
                                   }
                                   Ok::<_, std::convert::Infallible>(
                                       hyper::Response::new(http_body_util::Empty::<bytes::Bytes>::new()),
                                   )
                               }
                           }),
                       )
                       .await
                       .ok();
               }
           });

           tokio::spawn(async move {
               let stream = tokio::net::TcpStream::connect(addr).await.unwrap();
               let io = TokioIo::new(stream);
               let (mut sender, conn) = hyper::client::conn::http1::handshake(io).await.unwrap();
               tokio::spawn(async move {
                   let _ = conn.await;
               });
               let req = hyper::Request::builder()
                   .uri("http://localhost/")
                   .body(http_body_util::Empty::<bytes::Bytes>::new())
                   .unwrap();
               let _ = sender.send_request(req).await;
           });

           rx.await.unwrap()
       }

       #[tokio::test]
       async fn happy_path_ok_unit_redirects_303() {
           let mut req = make_request().await;
           let resp = ferro::__test_handle_action_result(
               Ok(()),
               "/dashboard",
               "test::happy_path",
               &mut req,
           );
           let r = unwrap_response(&resp);
           assert_eq!(r.status_code(), 303);
           // Back-compat query string default for success with no flash key.
           let loc = location_header(r).expect("Location header present");
           assert!(loc.starts_with("/dashboard?success="), "got: {loc}");
       }

       #[tokio::test]
       async fn success_override_redirect_and_flash() {
           let mut req = make_request().await;
           req.redirect_to("/dashboard/pagine/42");
           req.flash("created");
           let resp = ferro::__test_handle_action_result(
               Ok(()),
               "/dashboard",
               "test::success_override",
               &mut req,
           );
           let r = unwrap_response(&resp);
           let loc = location_header(r).expect("Location header present");
           assert!(loc.starts_with("/dashboard/pagine/42"), "got: {loc}");
           assert!(loc.contains("success=created"), "got: {loc}");
           // Flash payload should have been written to _action under success variant.
           // (Session inspection requires a live session middleware; if `session_mut`
           // returns None outside a request context, this assertion is dropped.
           // Document the limitation in the SUMMARY.)
       }

       #[tokio::test]
       async fn error_path_default_redirect_with_msg() {
           let mut req = make_request().await;
           let err = ActionError::msg("boom");
           let resp = ferro::__test_handle_action_result(
               Err(err),
               "/dashboard",
               "test::error_path",
               &mut req,
           );
           let r = unwrap_response(&resp);
           let loc = location_header(r).expect("Location header present");
           assert!(loc.starts_with("/dashboard?error=generic&msg="), "got: {loc}");
           assert!(loc.contains("boom"), "got: {loc}");
       }

       #[tokio::test]
       async fn error_path_with_redirect_override() {
           let mut req = make_request().await;
           let err = ActionError::unauthorized("login").redirect_to("/your-login-path");
           let resp = ferro::__test_handle_action_result(
               Err(err),

       }

       async fn t_180_02_open_redirect_error_side_falls_back() {
           let mut req = make_request().await;

               "/dashboard",
               "test::t_180_02_error",

       }

           assert!(!loc.contains("evil.example"), "got: {loc}");
       }

       #[tokio::test]
       async fn t_180_03_log_injection_sanitizer_strips_control_chars() {
           // We can't easily intercept tracing output in a unit test without
           // tracing-test, but we CAN assert the sanitizer behavior indirectly:
           // the query-string `msg` parameter receives the (percent-encoded) raw
           // message, while tracing receives the sanitized form. Since tracing
           // output is not asserted here, this test confirms only the message
           // round-trips to the URL unchanged. The sanitizer's unit-level
           // correctness is covered by the Plan 01 in-module test
           // `sanitize_strips_control_chars`.
           let mut req = make_request().await;
           let err = ActionError::msg("a\nfake-log-line");
           let resp = ferro::__test_handle_action_result(
               Err(err),
               "/dashboard",
               "test::t_180_03",
               &mut req,
           );
           let r = unwrap_response(&resp);
           let loc = location_header(r).expect("Location header present");
           // Newline is percent-encoded in the URL.
           assert!(loc.contains("%0A") || loc.contains("%0a"), "got: {loc}");
       }

       #[tokio::test]
       async fn warning_flash_variant_records_in_error_path() {
           let mut req = make_request().await;
           let err = ActionError::msg("careful").with_flash(FlashVariant::Warning);
           let resp = ferro::__test_handle_action_result(
               Err(err),
               "/dashboard",
               "test::warning",
               &mut req,
           );
           // The query string is identical regardless of FlashVariant; the
           // variant is reflected only in the session flash payload (which
           // requires a live session to inspect). At minimum the response
           // is a 303 redirect.
           let r = unwrap_response(&resp);
           assert_eq!(r.status_code(), 303);
       }
       ```
