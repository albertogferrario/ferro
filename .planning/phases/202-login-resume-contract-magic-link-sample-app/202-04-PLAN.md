    The full staged sequence (label each step with a comment mapping to SC-3):
    ```
    // SC-3 step 1: unauthenticated /authorize stored the return target
    store_oauth_return_to("/authorize?client_id=test&redirect_uri=...&...".to_string());

    // SC-3 step 2: request-link handler issued a token into the cache
    let token = "flow_token_unique";
    let user_id: i64 = 7;
    Cache::put(&format!("magic_link:{token}"), &user_id, Some(Duration::from_secs(15*60))).await.unwrap();
    assert!(Cache::get::<i64>(&format!("magic_link:{token}")).await.unwrap().is_some());

    // SC-3 step 3: verify consumes the token (single-use)
    let looked_up: Option<i64> = Cache::get(&format!("magic_link:{token}")).await.ok().flatten();
    let _ = Cache::forget(&format!("magic_link:{token}")).await;
    assert_eq!(looked_up, Some(user_id));
    assert!(Cache::get::<i64>(&format!("magic_link:{token}")).await.unwrap().is_none(), "token single-use");

    // SC-3 step 4: resume redirect targets the stored /authorize URL
    let resp = oauth_resume_redirect("/").unwrap(); // Ok branch
    assert_eq!(resp.status_code(), 302);            // confirm the exact status accessor from HttpResponse
    assert_eq!(/* Location header */, "/authorize?client_id=test&redirect_uri=...&...");
    ```
