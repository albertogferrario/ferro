       ```rust
       //! Bunny CDN purge adapter (feature `cdn-bunny`).
       use crate::cdn::PurgeApi;
       use crate::Error;
       use async_trait::async_trait;

       /// Configuration for the Bunny CDN URL-purge adapter.
       #[derive(Clone)]
       pub struct BunnyCdnConfig {
           /// CDN zone base URL, e.g. "https://myzone.b-cdn.net" (Bunny requires full URLs).
           pub cdn_base_url: String,
           /// Bunny API access key (BUNNY_ACCESS_KEY). Never logged.
           pub access_key: String,
       }
       impl std::fmt::Debug for BunnyCdnConfig {
           fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
               f.debug_struct("BunnyCdnConfig")
                   .field("cdn_base_url", &self.cdn_base_url)
                   .field("access_key", &"<redacted>")
                   .finish()
           }
       }
       impl BunnyCdnConfig {
           pub fn from_env() -> Self {
               Self {
                   cdn_base_url: std::env::var("BUNNY_CDN_URL").unwrap_or_default(),
                   access_key: std::env::var("BUNNY_ACCESS_KEY").unwrap_or_default(),
               }
           }
       }
       pub struct BunnyCdn { config: BunnyCdnConfig, client: reqwest::Client }
       impl BunnyCdn {
           pub fn new(config: BunnyCdnConfig) -> Self {
               Self { config, client: reqwest::Client::new() }
           }
       }
       #[async_trait]
       impl PurgeApi for BunnyCdn {
           async fn purge(&self, paths: &[String]) -> Result<(), Error> {
               if paths.is_empty() { return Ok(()); }
               if self.config.access_key.is_empty() {
                   return Err(Error::cdn("BUNNY_ACCESS_KEY not set"));
               }
               for path in paths {
                   let full_url = format!("{}/{}",
                       self.config.cdn_base_url.trim_end_matches('/'),
                       path.trim_start_matches('/'));
                   let resp = self.client
                       .post("https://api.bunny.net/purge")
                       .query(&[("url", full_url.as_str()), ("async", "false")])
                       .header("AccessKey", &self.config.access_key)
                       .send().await
                       .map_err(|e| Error::cdn(e.to_string()))?;
                   if !resp.status().is_success() {
                       let status = resp.status().as_u16();
                       let body = resp.text().await.unwrap_or_default();
                       return Err(Error::cdn(format!("Bunny purge status {status}: {body}")));
                   }
               }
               Ok(())
           }
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ferro-storage → Bunny API | Outbound HTTPS POST carrying the Bunny AccessKey header |
| ferro-storage → Cloudflare API | Outbound HTTPS POST carrying the CF bearer token |
| consumer → purge(paths) | Semi-trusted relative paths placed into query/JSON |
| env → Bunny/CF config from_env | BUNNY_ACCESS_KEY / CF_API_TOKEN read from process env |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-188-10 | Information Disclosure | BUNNY_ACCESS_KEY / CF_API_TOKEN in Debug/logs | mitigate | Both configs use a hand-written Debug redacting the secret as "<redacted>" (no derive Debug); grep-asserted. No tracing line prints the secret. |
| T-188-11 | Tampering / Injection | consumer-supplied paths into Bunny query / CF JSON | mitigate | Bunny: paths go through reqwest `.query()` (URL-encoded by reqwest). CF: paths go into a serde_json body (escaped). No manual string interpolation into the request line. |
| T-188-12 | Spoofing / SSRF | adapter endpoint host | mitigate | Bunny/CF hosts are fixed string constants (`api.bunny.net`, `api.cloudflare.com`); zone_id/cdn_base come from trusted env. No untrusted host injection. |
| T-188-13 | Info Disclosure (transport) | secrets over the wire | mitigate | reqwest rustls-tls (Plan 01); HTTPS only; cert verification at default (never disabled). |

block-on-high: the only high-severity item (secret leakage T-188-10) is mitigated with grep-asserted redaction. No unmitigated high severity → gate passes.
</threat_model>
