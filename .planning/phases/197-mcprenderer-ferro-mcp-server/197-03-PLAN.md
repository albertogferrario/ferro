```rust
use sea_orm::{ConnectionTrait, DatabaseBackend, Statement};

pub async fn dispatch(
    service: &ServiceDef,
    filters: serde_json::Value,
    limit: u64,
    offset: u64,
    db: &sea_orm::DatabaseConnection,
) -> crate::Result<DispatchResult> {
    let backend = db.get_database_backend();
    let table = format!("{}s", service.name.to_lowercase()); // TODO: ServiceDef.table

    let mut where_clauses = Vec::new();
    let mut values: Vec<sea_orm::Value> = Vec::new();
    let mut idx = 1usize;
    if let Some(obj) = filters.as_object() {
        for (key, val) in obj {
            // ALLOWLIST: key must be a known field
            if !service.fields.iter().any(|f| &f.name == key) {
                return Err(crate::Error::Database(format!("unknown filter field: {key}")));
            }
            where_clauses.push(format!("\"{}\" = {}", key, placeholder(backend, idx)));
            values.push(json_to_sea_value(val));
            idx += 1;
        }
    }
    let where_str = if where_clauses.is_empty() { String::new() }
        else { format!(" WHERE {}", where_clauses.join(" AND ")) };

    // COUNT
    let count_sql = format!("SELECT COUNT(*) as cnt FROM \"{table}\"{where_str}");
    let count_stmt = Statement::from_sql_and_values(backend, &count_sql, values.clone());
    let total: u64 = db.query_one(count_stmt).await
        .map_err(|e| crate::Error::Database(e.to_string()))?
        .and_then(|r| r.try_get_by::<i64, _>("cnt").ok()).unwrap_or(0) as u64;

    // DATA + pagination (bind limit/offset as params too)
    let data_sql = format!(
        "SELECT * FROM \"{table}\"{where_str} LIMIT {} OFFSET {}",
        placeholder(backend, idx), placeholder(backend, idx + 1)
    );
    values.push(sea_orm::Value::BigInt(Some(limit as i64)));
    values.push(sea_orm::Value::BigInt(Some(offset as i64)));
    let data_stmt = Statement::from_sql_and_values(backend, &data_sql, values);
    let rows = db.query_all(data_stmt).await
        .map_err(|e| crate::Error::Database(e.to_string()))?;

    Ok(DispatchResult { rows: rows_to_json(rows), total, limit, offset })
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| agent-controlled `filters` payload → dispatch SQL | The `filters` JSON object is the only agent-controlled input that reaches SQL construction. Filter KEYS would become column names; VALUES would become predicate operands. This is the single real security boundary in Phase 197. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-197-06 | Tampering / Elevation (SQL injection via filter KEYS) | `dispatch()` WHERE construction | mitigate | Filter keys are allow-listed against `service.fields[].name` before any SQL string assembly; an unknown key returns `Err`, never interpolated. Acceptance criterion: the `{"bogus":1}` integration test asserts `is_err()`. Column names originate only from the trusted `ServiceDef`, never raw from the payload. |
| T-197-07 | Tampering (SQL injection via filter VALUES) | `dispatch()` value binding | mitigate | All filter values + limit/offset are bound via `Statement::from_sql_and_values` parameters (`json_to_sea_value` → `sea_orm::Value`); no value is string-interpolated into SQL. |
| T-197-08 | Tampering (table-name injection) | table-name derivation | accept | Table name is derived from `service.name` (developer-controlled at projection-definition time, snake_case by builder convention), not from the call payload. Still double-quoted in SQL. Risk is developer self-injection only; acceptable for the skeleton. TODO records the future `ServiceDef.table` field. |
| T-197-09 | Elevation (cross-tenant data access) | dispatch returns all rows | accept (this phase) | Phase 197 is in-process, no auth, no tenant scoping by design (deferred to Phase 200). The signature intentionally carries NO ownership filter to avoid a duplicate control surface competing with the Phase 200 policy layer. Documented as deferred in ROADMAP/CONTEXT. |
</threat_model>
