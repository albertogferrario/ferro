    ```rust
    /// Errors from the race-safe resolve helper.
    #[derive(Debug, thiserror::Error)]
    pub enum ResolveError {
        #[error("projection read failed: {0}")]
        Projection(#[from] ProjectionError),
        #[error("broadcast subscribe failed: {0}")]
        Broadcast(String),
        #[error("resolve channel closed before a result arrived")]
        ChannelClosed,
        #[error("resolve timed out")]
        Timeout,
    }

    /// Resolve an offloaded result race-safely (D-09): subscribe FIRST, read the snapshot back
    /// ONCE (catches an already-completed/failed handle), then await the delta and read the
    /// authoritative snapshot on wake.
    ///
    /// The delta is a wakeup + client-safe payload; the snapshot is authoritative (D-06). This
    /// server-side helper reads the FULL envelope (`read_result`, raw error) — a browser uses the
    /// redacted read-back (`read_result_redacted`) instead. Because it subscribes before reading
    /// back, it cannot miss a delta that fires between the read-back and the await (Pitfall 3).
    ///
    /// `timeout: None` waits until the channel closes or a result arrives; a terminally failed
    /// job records a `failed` snapshot + delta (Plan 02), so the only unbounded wait is a job
    /// that never runs — pass `Some(dur)` to bound it.
    pub async fn resolve<T: OffloadSerializable>(
        handle: &ferro_queue::OffloadHandle<T>,
        broadcaster: &Arc<ferro_broadcast::Broadcaster>,
        db: &DatabaseConnection,
        timeout: Option<std::time::Duration>,
    ) -> Result<OffloadResult<T>, ResolveError> {
        use ferro_broadcast::ServerMessage;
        let key = handle.key();
        let channel = format!("projection.{}.{}", OFFLOAD_PROJECTION_NAME, key);
        let client_id = format!("{}-resolve-{}", key, uuid::Uuid::new_v4());

        // 1. Subscribe FIRST (Pitfall 3).
        let (tx, mut rx) = tokio::sync::mpsc::channel::<ServerMessage>(8);
        broadcaster.add_client(client_id.clone(), tx);
        broadcaster
            .subscribe(&client_id, &channel, None, None)
            .await
            .map_err(|e| ResolveError::Broadcast(e.to_string()))?;

        // 2. Read back ONCE — short-circuit an already-terminal handle.
        if let Some(result) = read_result::<T>(key, db).await? {
            if !matches!(result, OffloadResult::Pending) {
                broadcaster.remove_client(&client_id);
                return Ok(result);
            }
        }

        // 3. Await the delta, then read the authoritative snapshot on wake.
        let wait = async {
            while let Some(msg) = rx.recv().await {
                if let ServerMessage::Event(b) = msg {
                    if b.event == "offload.result" {
                        break;
                    }
                }
            }
            read_result::<T>(key, db)
                .await
                .map(|opt| opt.unwrap_or(OffloadResult::Pending))
                .map_err(ResolveError::from)
        };
        let out = match timeout {
            Some(d) => tokio::time::timeout(d, wait)
                .await
                .map_err(|_| ResolveError::Timeout)?,
            None => wait.await,
        };
        broadcaster.remove_client(&client_id);
        out
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| resolve() subscriber → broadcast channel | The resolve helper subscribes to the public capability-keyed channel and treats the delta as an untrusted wakeup, reconciling against the authoritative snapshot. |
| documented client → redacted delta / redacted read-back | The documented browser path consumes only redacted surfaces (`read_result_redacted` + the redacted delta), never the raw error. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-247-info-disclosure | Information Disclosure | `offload_failed_delta_is_redacted` test / documented client path | mitigate | The test concretely asserts `data.get("error").is_none()` on the failed delta while the raw error remains only in the `read_result` snapshot; the docs direct browser clients to `read_result_redacted`. This is the concrete verification the security gate requires. |
| T-247-hostile-payload | Tampering | live-redis bus (`redis_cross_replica`) | mitigate | Inherited from 246.1 T-246.1-03: strict `serde_json` bus-envelope parse, drop on error. The redis test rides the shipped `RedisTransport`; Phase 247 adds no new parse surface. |
| T-247-resolve-wakeup | Spoofing | resolve() delta-triggered read-back | mitigate | resolve() treats the delta only as a wakeup and reads the authoritative snapshot for the result (D-06), so a spurious/forged `offload.result` event on the channel cannot inject a fabricated value — it only triggers a snapshot read keyed by the handle the caller already holds. |
| T-247-handle-enum | Spoofing | documented public channel | accept | D-11 capability model (unguessable UUID v4); documented accepted caveat, restated in queues.md. No new control. |
</threat_model>
