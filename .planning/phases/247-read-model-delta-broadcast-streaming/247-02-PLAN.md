           broadcaster: Arc<ferro_broadcast::Broadcaster>,
       ) {
           let _ = OFFLOAD_BROADCASTER.set(broadcaster);
           ferro_queue::register_offload_result_hook(|key, outcome, db| {
               Box::pin(async move {
                   // Build the client-facing (redacted) delta payload BEFORE consuming `outcome`.

                       Ok(value) => serde_json::json!({ "status": "completed", "value": value }),
                       Err(_) => serde_json::json!({ "status": "failed" }), // D-05: no raw error
                   };

                   let res = match outcome {
                       Ok(value) => persist_result_raw(&key, value, db).await,
                       Err(msg) => persist_error(&key, &msg, db).await, // raw error stays in snapshot

                   if let Err(e) = res {

                   }

                       broadcast_delta(b, &key, delta).await;
                   }
               })
           });

    ///
    /// The request-side entrypoint: dispatches the job via `Offloadable::offload()`, then writes
    /// `{"status":"pending"}` under the returned handle key so a read-back can tell an unknown
    /// handle (no snapshot) from work in flight (pending). The pending write is on the framework

    /// caller subscribes on. Does NOT await the result — the request stays non-blocking (SC#2).
    ///
    /// A pending-write failure is logged at `warn!` and swallowed: the job is already enqueued,
    /// so a missing pending row degrades a read-back to `None` (unknown) rather than corrupting
    /// state; the result snapshot still lands when the worker finishes.
    pub async fn enqueue_and_mark_pending<J>(
        job: J,
        db: &DatabaseConnection,
    ) -> Result<ferro_queue::OffloadHandle<J::Output>, ferro_queue::Error>

        J: ferro_queue::Offloadable,

        let handle = job.offload().await?;
        if let Err(e) = persist_pending(handle.key(), db).await {
            tracing::warn!(handle_key = %handle.key(), error = %e,
                "offload pending marker write failed; result path unaffected");
        }
        Ok(handle)
    }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| worker → broadcast bus → subscribed client | The worker publishes a delta that crosses the (possibly cross-replica, 246.1) broadcast transport to any client subscribed on the public capability-keyed channel. |
| worker error string → broadcast payload | The raw `Display` error must not cross into the client-facing delta. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-247-info-disclosure | Information Disclosure | result hook broadcast payload (framework/src/offload.rs) | mitigate | D-05: the failed delta arm builds `{"status":"failed"}` with NO `error` field; the raw error is written only to the snapshot via `persist_error`. Acceptance criterion asserts the `Err` delta arm contains no `error` key; Plan 03's integration test asserts `data["error"]` is absent for a failed run. |
| T-247-handle-enum | Spoofing | public channel `projection.offload.result.{handle}` | accept | D-11 capability model: the handle is an unguessable UUID v4 (122 bits) minted server-side (Phase 245), functioning as a capability token. Documented accepted caveat (a leaked handle exposes one result); no new control this phase. |
| T-247-hostile-payload | Tampering | cross-replica bus payload (Redis path, 246.1) | mitigate | Inherited from 246.1 T-246.1-03: the transport parses the bus envelope with strict `serde_json` and drops on parse error. Phase 247 adds no new bus-parsing surface; the delta rides the existing `ServerMessage::Event` fan-out. |
| T-247-hook-failfail | Denial of Service | result hook / job retry | mitigate | Pitfall 5 / D-02: the hook swallows persist and broadcast errors (logs `warn!`, returns `()`), so a transient broadcast failure cannot mark the job failed and trigger a retry storm. |
