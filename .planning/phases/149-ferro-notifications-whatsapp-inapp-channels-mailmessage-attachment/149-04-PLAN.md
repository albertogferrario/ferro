```rust
// Body part — single-part for backward-compat when no attachments,
// SinglePart wrapped in MultiPart::mixed otherwise.
use lettre::message::{Attachment, MultiPart, SinglePart};

let email = if message.attachments.is_empty() {
    // Backward-compatible single-part path
    if let Some(ref html) = message.html {
        email_builder
            .header(ContentType::TEXT_HTML)
            .body(html.clone())
            .map_err(|e| Error::mail(format!("Failed to build email: {e}")))?
    } else {
        email_builder
            .header(ContentType::TEXT_PLAIN)
            .body(message.body.clone())
            .map_err(|e| Error::mail(format!("Failed to build email: {e}")))?
    }
} else {
    // Multipart path — body becomes a SinglePart inside MultiPart::mixed
    let body_part = if let Some(ref html) = message.html {
        SinglePart::html(html.clone())
    } else {
        SinglePart::plain(message.body.clone())
    };

    let mut mp = MultiPart::mixed().singlepart(body_part);
    for att in &message.attachments {
        let ct = ContentType::parse(&att.content_type)
            .map_err(|e| Error::mail(format!("Invalid content-type '{}': {e}", att.content_type)))?;
        let part = Attachment::new(att.filename.clone())
            .body(att.content.clone(), ct);
        mp = mp.singlepart(part);
    }

    email_builder
        .multipart(mp)
        .map_err(|e| Error::mail(format!("Failed to build multipart email: {e}")))?
};
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| `MailAttachment.filename` / `content_type` → SMTP wire format | Caller-supplied strings; lettre owns header escaping. |
| `MailAttachment.content` → Resend HTTP API | Bytes are base64-encoded then JSON-serialized; standard alphabet ensures Resend round-trips without corruption. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-149-W2-01 | Tampering / Header Injection | filename and content_type passed to lettre | mitigate | Use `Attachment::new(filename).body(content, ContentType::parse(content_type)?)` — lettre handles all header escaping. `ContentType::parse` is fallible and is mapped to `Error::Mail` (not propagated as a string concat into headers). |
| T-149-W2-02 | Information Disclosure | Resend payload includes attachment content as base64 in plaintext JSON | accept | TLS-only transport via reqwest's default. Same trust as the existing `body` and `subject` fields in the payload. |
| T-149-W2-03 | Tampering | Base64 alphabet must match Resend's expectation | mitigate | Test `test_base64_encoding_uses_standard_alphabet` locks the standard alphabet. URL-safe substitution would corrupt binary content. |
| T-149-W2-04 | Resource Exhaustion (Memory) | base64 encoding allocates ~4/3× the size of the original bytes | accept | Bounded by the 25 MB per-attachment cap from Plan 03. Worst case ~33 MB transient string per attachment — within standard server memory budgets. |

No HIGH severity threats. Plan is not blocked.
