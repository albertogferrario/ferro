    ```rust
    #[cfg(test)]
    mod tests {
        use schemars::schema_for;
        use serde_json::Value;

        #[test]
        fn schema_probe_field_meaning_any_of_shape() {
            use ferro_projections::FieldMeaning;
            let schema: Value = serde_json::to_value(schema_for!(FieldMeaning)).unwrap();
            let any_of = schema["anyOf"].as_array().expect("FieldMeaning must emit anyOf");
            assert!(any_of.len() >= 2, "expected >=2 anyOf branches, got {}", any_of.len());
            assert_eq!(any_of[0]["type"], "string");
            let variants = any_of[0]["enum"].as_array().expect("first branch must be closed enum");
            assert!(variants.iter().any(|v| v == "money"), "known variants must include money");
            assert_eq!(any_of[1]["type"], "string");
            assert!(any_of[1].get("enum").is_none(), "second branch must be open string (the Custom escape hatch)");
        }

        #[test]
        fn schema_probe_intent_any_of_shape() {
            use ferro_projections::Intent;
            let schema: Value = serde_json::to_value(schema_for!(Intent)).unwrap();
            let any_of = schema["anyOf"].as_array().expect("Intent must emit anyOf");
            assert!(any_of.len() >= 2);
            let variants = any_of[0]["enum"].as_array().expect("first branch must be closed enum");
            let names: Vec<&str> = variants.iter().filter_map(|v| v.as_str()).collect();
            for expected in ["browse","focus","collect","process","summarize","analyze","track"] {
                assert!(names.contains(&expected), "Intent enum missing {expected}");
            }
            assert!(any_of[1].get("enum").is_none(), "second branch must be open string");
        }
    }
    ```

The RESEARCH examples wrote `.to_value()` — confirm the real 1.2.0 API and use it).
