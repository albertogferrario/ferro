    ```rust
    use std::collections::HashSet;
    use serde_json::{Map, Value};

    /// Normalize a schemars 1.x schema for Anthropic / OpenAI structured-output APIs.
    ///
    /// Order is mandatory (see RESEARCH Pitfall 2): projection-enum closing (Plan 03) runs
    /// on `$defs` BEFORE ref inlining. This generic path: (1) inline all `$ref` recursively
    /// with a cycle guard, (2) drop `$defs`/`definitions`, (3) strip Anthropic-rejected
    /// keywords and add `additionalProperties: false` to every object-with-properties.
    pub fn for_structured_output(schema: Value) -> Value { /* ... */ }

    fn resolve_refs(node: Value, defs: &Map<String, Value>, visited: &mut HashSet<String>) -> Value { /* ... */ }

    /// Strip rejected keywords + add additionalProperties:false. Recursive rebuild (Pitfall 3).
    fn normalize_node(node: Value) -> Value { /* ... */ }
    ```

    Implementation rules:
    - `for_structured_output`: clone `$defs` (and `definitions`) into a `defs` map; call `resolve_refs(root, &defs, &mut HashSet::new())`; remove `$defs`/`definitions` from the resulting root object; then `normalize_node(root)`.
    - `resolve_refs`: match on `Value`. If object is exactly `{"$ref": s}` where `s` strips the `#/$defs/` or `#/definitions/` prefix to a name: if name already in `visited` → return `json!({"type":"object"})`; else insert name, recurse into `defs[name].clone()`, remove name from visited on the way out. For other objects: rebuild a new `Map` recursing each value. For arrays: rebuild recursing each element. Else: return unchanged.
    - `normalize_node`: define `const STRIP_KEYWORDS: &[&str] = &["$schema","$id","title","examples","minimum","maximum","multipleOf","minLength","maxLength","pattern"];`. For objects: rebuild dropping any key in STRIP_KEYWORDS; for `format`, drop it only if its string value is NOT one of the supported string formats (`date-time`,`date`,`time`,`duration`,`email`,`hostname`,`uri`,`ipv4`,`ipv6`,`uuid`); recurse into remaining values; AFTER rebuild, if the node has `type=="object"` AND contains `properties`, insert `additionalProperties: false`. For arrays: recurse elements. Else: unchanged.
    - No `unwrap()` in production paths — use `as_object()`/`as_str()` with graceful fallthrough.
