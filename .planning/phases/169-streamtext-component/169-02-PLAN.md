/// Dependency-free inline EventSource wiring for `StreamText` components.
/// Appends streamed tokens as text nodes (never `innerHTML`), removes the
/// placeholder on the first token, and closes the source on `event: done`
/// to prevent `EventSource` auto-reconnect. Emitted at most once per page.
const FERRO_STREAM_TEXT_INIT: &str = r#"(function(){
  document.querySelectorAll('[data-ferro-stream-url]').forEach(function(el){
    var src = new EventSource(el.dataset.ferroStreamUrl);
    var placeholder = el.querySelector('[data-ferro-stream-placeholder]');
    var loading = el.querySelector('[data-ferro-stream-loading]');
    var firstToken = true;
    src.onmessage = function(e){
      if(firstToken){ firstToken=false; if(placeholder) placeholder.remove(); }
      el.appendChild(document.createTextNode(e.data));
    };
    src.addEventListener('done', function(){
      src.close();
      if(loading) loading.remove();
    });
    src.onerror = function(){
      src.close();
      if(loading) loading.remove();
    };
  });
})();"#;

/// Returns the StreamText EventSource init script if the spec contains at least
/// one `StreamText` element; otherwise an empty `Vec`. Walks `spec.elements`
/// the same way `collect_plugin_types` does. Returns at most one entry so the
/// script is emitted exactly once regardless of how many StreamText elements
/// the spec contains.
fn collect_builtin_init_scripts(spec: &Spec) -> Vec<String> {
    let has_stream_text = spec
        .elements
        .values()
        .any(|el| el.type_name == "StreamText");
    if has_stream_text {
        vec![FERRO_STREAM_TEXT_INIT.to_string()]
    } else {
        vec![]
    }
}
```

pub fn render_spec_to_html_with_plugins(spec: &Spec, data: &Value) -> RenderResult {
    let html = render_spec_to_html(spec, data);
    let builtin_scripts = collect_builtin_init_scripts(spec);
    let plugin_types = collect_plugin_types(spec);
    if plugin_types.is_empty() && builtin_scripts.is_empty() {
        return RenderResult {
            html,
            css_head: String::new(),
            scripts: String::new(),
        };
    }
    let type_names: Vec<String> = plugin_types.into_iter().collect();
    let assets = collect_plugin_assets(&type_names);
    let all_init_scripts: Vec<String> = assets
        .init_scripts
        .iter()
        .chain(builtin_scripts.iter())
        .cloned()
        .collect();
    RenderResult {
        html,
        css_head: render_css_tags(&assets.css),
        scripts: render_js_tags(&assets.js, &all_init_scripts),
    }
}
```

Note: when `plugin_types` is empty but `builtin_scripts` is non-empty, `collect_plugin_assets`
is called with an empty `type_names` slice and returns empty CSS/JS/plugin-init — that is
correct; the merged `all_init_scripts` still carries the builtin script.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| SSE server stream → browser DOM | Token payloads (`e.data`) arrive over the SSE connection from an LLM/server stream — untrusted content that the inline init script writes into the DOM. |
| spec author → emitted page JS | The init script is emitted whenever a spec contains a StreamText element; it is a fixed constant (not parameterized by spec values), so the only injection surface is the token stream at runtime. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-169-02 | Tampering (XSS via streamed token injection) | `FERRO_STREAM_TEXT_INIT` inline JS — token append path | mitigate | Tokens appended via `document.createTextNode(e.data)` + `appendChild`, never `innerHTML`/`insertAdjacentHTML` (Task 2). Acceptance: Task 3 test asserts emitted script contains `createTextNode` and does NOT contain `innerHTML`; `grep -c "innerHTML" render/mod.rs` == 0. **High severity — this is the core security property of the component.** |
| T-169-03 | Tampering (XSS, same vector, distinct assertion) | inline JS — DOM write API selection | mitigate | Same text-node mechanism; asserted by the same `!scripts.contains("innerHTML")` check (Task 3). The two threat IDs share one mitigation: HTML is never parsed from token data. |
| T-169-04 | Denial of Service (EventSource reconnect loop) | inline JS lifecycle | mitigate | `src.close()` on `event: done` and on `onerror` (Task 2). Acceptance: Task 3 asserts the script contains `'done'` + `close()`. Documented server contract (`event: done` required) is authored in Plan 03. |

## Out-of-scope / accepted

| Threat | Disposition | Rationale |
|--------|-------------|-----------|
| Init script unconditionally querying all `[data-ferro-stream-url]` elements on the page | accept | The selector + attribute name is a fixed framework convention, not attacker-controllable; a spec author placing StreamText is the intended trigger. No PII, no auth boundary crossed. |
