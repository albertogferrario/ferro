  ```rust
  #[test]
  fn switch_at_depth_8_renders_role_switch() {
      // Consumer's staff-detail nesting:
      // dashboard (depth 1) → root layout (2) → DetailPage (3) → Tabs (4)
      // → Card (5) → Form (6) → Grid (7, "row") → Switch (8)
      //
      // Before Phase 175-01 (F1): MAX_NESTING_DEPTH = 5 → the walker tripwire fired
      // at depth 7+, emitting the "depth limit exceeded" comment instead of dispatching
      // to render_switch. The consumer's Switch was invisibly stripped.
      //
      // After Phase 175-01: MAX_NESTING_DEPTH = 16 → depth 8 is well within the
      // limit, dispatch reaches render_switch, and the role="switch" markup emerges.
      //
      // This test pins the F1+F4 closure: if a future regression lowers the depth
      // limit below 8 or breaks the Switch dispatch, this test fires.

      use crate::spec::{Element, Spec, SCHEMA_VERSION};
      use std::collections::HashMap;

      // Helper: a Grid container whose only child is the next-level element.
      fn grid_with_child(id: &str, child_id: &str) -> Element {
          Element {
              type_name: "Grid".to_string(),
              props: serde_json::json!({ "columns": 1 }),
              children: vec![child_id.to_string()],
              visible: None,
              footer: None,
              action: None,
              each: None,
              if_: None,
          }
      }

      let mut elements = HashMap::new();
      // Chain seven Grid containers (depths 1..=7) ending in a Switch at depth 8.
      for i in 1..=7 {
          elements.insert(
              format!("g{}", i),
              grid_with_child(&format!("g{}", i), &format!("g{}", i + 1)),
          );
      }
      // Replace the last Grid (g7) with one whose child is the Switch leaf.
      elements.insert(
          "g7".to_string(),
          grid_with_child("g7", "leaf"),
      );
      elements.insert(
          "leaf".to_string(),
          Element {
              type_name: "Switch".to_string(),
              props: serde_json::json!({
                  "field": "day_1_is_open",
                  "label": "Aperto"
              }),
              children: vec![],
              visible: None,
              footer: None,
              action: None,
              each: None,
              if_: None,
          },
      );
      let spec = Spec {
          schema: SCHEMA_VERSION.to_string(),
          root: "g1".to_string(),
          elements,
          title: None,
          layout: None,
          data: serde_json::Value::Null,
      };

      // Must parse — depth 8 is within MAX_NESTING_DEPTH = 16.
      let json = serde_json::to_value(&spec).unwrap();
      let parsed = Spec::from_json(json).expect("depth-8 spec must parse after Phase 175-01");

      let html = crate::render::render_spec_to_html(&parsed, &serde_json::Value::Null);

      assert!(
          html.contains("role=\"switch\""),
          "Switch at depth 8 must render role=\"switch\"; got: {}",
          html
      );
      assert!(
          !html.contains("depth limit exceeded"),
          "Switch at depth 8 must not be stripped by the walker tripwire; got: {}",
          html
      );
      assert!(
          !html.contains("cycle guard tripped"),
          "Walker tripwire must not emit the deprecated cycle-guard diagnostic; got: {}",
          html
      );
  }
  ```
