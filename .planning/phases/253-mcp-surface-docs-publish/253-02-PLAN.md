```rust
use ferro_json_ui::component::{Size, Tone, Variant};
use ferro_json_ui::design::{rules as design_rules, KNOWN_INTENTS};
use strum::VariantArray;

let variant_values: Vec<String> = Variant::VARIANTS.iter().map(|v| v.as_ref().to_string()).collect();
let tone_values: Vec<String> = Tone::VARIANTS.iter().map(|v| v.as_ref().to_string()).collect();
let size_values: Vec<String> = Size::VARIANTS.iter().map(|v| v.as_ref().to_string()).collect();

let mut intent_rules: std::collections::HashMap<String, Vec<DesignRuleRef>> =
    KNOWN_INTENTS.iter().map(|&i| (i.to_string(), Vec::new())).collect();
intent_rules.insert("all".to_string(), Vec::new()); // rules with empty intents apply everywhere
for rule in design_rules() {
    let r = DesignRuleRef { id: rule.id, title: rule.title, rationale: rule.rationale };
    if rule.intents.is_empty() {
        intent_rules.entry("all".to_string()).or_default().push(r);
    } else {
        for &intent in rule.intents {
            intent_rules.entry(intent.to_string()).or_default().push(DesignRuleRef {
                id: rule.id, title: rule.title, rationale: rule.rationale,
            });
        }
    }
}
```
