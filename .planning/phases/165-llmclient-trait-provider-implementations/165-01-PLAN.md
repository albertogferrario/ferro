1. Create `ferro-ai/src/client/mod.rs`. Declare the three submodules at the top:

   Then define exactly (doc-comment every public symbol — workspace convention):
   ```rust
   use crate::error::Error;
   use async_trait::async_trait;
   use futures::stream::BoxStream;

   /// Opaque stream of text tokens from a streaming LLM completion.
   pub type TokenStream = BoxStream<'static, Result<String, Error>>;

   /// Role of a message in a completion request.
   pub enum Role { User, Assistant }

   /// A single message in a completion request.
   pub struct Message { pub role: Role, pub content: String }

   /// Request for a text completion.
   pub struct CompletionRequest {
       /// Optional system prompt.
       pub system: Option<String>,
       /// Conversation messages in order.
       pub messages: Vec<Message>,
       /// Maximum tokens in the response.
       pub max_tokens: u32,
       /// Optional per-request model override; None resolves to the client's default_model().
       pub model_override: Option<String>,
       /// Optional JSON schema for structured output, passed through to the provider
       /// (Phase 166 normalizer handles schemars -> provider-specific format).
       pub schema: Option<serde_json::Value>,
   }

   /// Provider-agnostic LLM client.
   #[async_trait]
   pub trait LlmClient: Send + Sync {
       /// The provider's default model identifier.
       fn default_model(&self) -> &str;
       /// Run a non-streaming completion, returning the full text.
       async fn complete(&self, request: CompletionRequest) -> Result<String, Error>;
       /// Run a streaming completion, returning a token stream.
       async fn complete_stream(&self, request: CompletionRequest) -> Result<TokenStream, Error>;
       /// Generate an embedding. Returns Err(Error::Unsupported) for providers without an embeddings endpoint.
       async fn embed(&self, text: &str) -> Result<Vec<f32>, Error>;
   }
   ```
   `model_override` is included now because PATTERNS classifier-bridge (lines 476-482) and RESEARCH build_body
