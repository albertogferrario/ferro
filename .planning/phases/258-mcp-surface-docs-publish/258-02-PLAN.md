<!-- Props ground truth (verified: ferro-json-ui/src/component.rs:1412–1529). Copy these EXACTLY. -->

TileGrid props:
  data_path: string (req) — JSON pointer to items array iterated via $each; emitted context for the grid
  form_id: string (req) — HTML id of the Form owning hidden inputs; emitted as data-selection-form
  categories_path: string|null — JSON pointer to string array for the integrated category strip; absent → no strip
  columns: number|null — base viewport column count (default: 2)
  search: boolean|null — enable client-side text search input
  search_placeholder: string|null — placeholder (default: "Search"); ignored when search absent/false
  all_label: string|null — "Show all" tab label for the integrated strip (default: "All"); ignored when categories_path absent

SelectionPanel props:
  form_id: string (req) — must match the paired TileGrid form_id; emitted as data-selection-form
  empty_message: string|null — placeholder text when panel has no line items
  currency: string|null — currency symbol (e.g. "€") prepended to the integer-cents total; no locale tables
  total_label: string|null — running-total row label (default: "Total")

FilterTabs props:
  items: string[] (default []) — category labels rendered as filter tabs; may be $data-bound
  all_label: string|null — "Show all" tab label (default: "All")
  (zero-prop FilterTabs renders an All-only strip)

QuantityStepper props:
  field: string (req) — name of the hidden input this stepper drives via data-qty-inc/data-qty-dec
  min: number|null — lower bound (default: 0)
  max: number|null — upper bound; unbounded when absent
  step: number|null — increment size (default: 1)

Numpad props:
  target_field: string (req) — name of the input this numpad writes into; emitted as data-numpad-target
  mode: "quantity" | "price" (default: "quantity") — quantity = integer entry; price = two-decimal monetary entry
