    Replace the stub in ferro-cli/src/commands/do_init.rs with full implementation:

    ```rust
    #[derive(clap::Args, Debug)]
    pub struct DoInitArgs {
        #[arg(long)]
        pub force: bool,
    }

    pub fn run(args: DoInitArgs) -> anyhow::Result<()> {
        let root = crate::project::walk_up_to_project_root()?;
        let package_name = crate::project::read_package_name(&root)?;
        let name = crate::templates::do_::sanitize_do_app_name(&package_name);

        // Auto-detect repo
        let repo = std::process::Command::new("git")
            .args(["remote", "get-url", "origin"])
            .current_dir(&root)
            .output()
            .ok()
            .and_then(|o| String::from_utf8(o.stdout).ok())
            .and_then(|s| crate::templates::do_::parse_git_remote(s.trim()))
            .unwrap_or_else(|| "owner/your-repo".to_string());

        // Workers
        let bins = crate::templates::docker::read_bins(&root)?;
        let web_bin = bins.iter().find(|b| **b == name || **b == package_name).cloned().unwrap_or_else(|| name.clone());
        let workers: Vec<String> = bins.iter()
            .filter(|b| **b != web_bin)
            .filter(|b| !crate::templates::do_::is_test_like_bin(b))
            .cloned()
            .collect();

        // Env keys — REQUIRED file
        let env_path = root.join(".env.production");
        if !env_path.exists() {
            anyhow::bail!(
                ".env.production not found.\n\
                 Create it from .env.example and fill in production values:\n  \
                 cp .env.example .env.production\n  \
                 $EDITOR .env.production"
            );
        }
        let env_keys = crate::deploy::env_production::read_env_production_keys(&env_path)?;

        let ctx = crate::templates::do_::AppYamlContext {
            name,
            repo,
            web_bin,
            workers,
            env_keys,
        };
        let yaml = crate::templates::do_::render_app_yaml(&ctx);

        let target = root.join(".do/app.yaml");
        std::fs::create_dir_all(target.parent().unwrap())?;
        write_if_absent_or_force(&target, &yaml, args.force)?;

        Ok(())
    }
    ```

    Important per SCOPE §7: do NOT call any ci.yml renderer. do NOT write `.github/workflows/ci.yml`. The ci:init command remains separate.

    If the existing project.rs doesn't have `read_package_name`, add it or reuse the existing helper.

    Note: the module name `do` conflicts with the Rust keyword, so use `do_` or similar as the Rust module ident even though the file is `do.rs`. Match whatever convention templates/mod.rs already uses.
