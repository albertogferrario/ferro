    ```rust
    #[derive(clap::Args, Debug)]
    pub struct DockerInitArgs {
        #[arg(long)]
        pub force: bool,
        /// One-shot ferro version override (does not mutate Cargo.toml).
        #[arg(long)]
        pub ferro_version: Option<String>,
    }

    pub fn run(args: DockerInitArgs) -> anyhow::Result<()> {
        let root = crate::project::walk_up_to_project_root()?;
        let metadata = crate::project::read_deploy_metadata(&root)?;
        let rust_channel = crate::templates::docker::read_rust_channel(&root);
        let bins = crate::templates::docker::read_bins(&root)?;
        let has_frontend = root.join("frontend/package.json").exists();
        let copy_dirs_present: Vec<String> = metadata.copy_dirs.iter()
            .filter(|d| root.join(d).exists())
            .cloned()
            .collect();

        let ctx = crate::templates::docker::DockerContext {
            rust_channel,
            has_frontend,
            bins,
            copy_dirs_present,
            runtime_apt: metadata.runtime_apt.clone(),
        };
        let dockerfile = crate::templates::docker::render_dockerfile(&ctx);

        // Respect --force for existing files.
        write_if_absent_or_force(&root.join("Dockerfile"), &dockerfile, args.force)?;
        write_if_absent_or_force(
            &root.join(".dockerignore"),
            include_str!("../templates/files/docker/dockerignore.tpl"),
            args.force,
        )?;

        // Version override precedence: CLI flag → metadata.ferro_version → path-dep workspace version → "*"
        let override_str = args.ferro_version.as_deref().or(metadata.ferro_version.as_deref());
        crate::deploy::rewrite_ferro_version::rewrite_cargo_docker_toml(&root, override_str)?;

        Ok(())
    }
    ```

    Implement `write_if_absent_or_force` as a small helper (private module-local fn). Use the existing `walk_up_to_project_root` from project.rs.
