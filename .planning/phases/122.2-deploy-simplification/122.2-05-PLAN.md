    use std::path::Path;

    /// Read .env.production and yield the list of keys.
    /// Per SCOPE §6: no value parsing, no comment stripping, no classification.
    pub fn read_env_production_keys(path: &Path) -> anyhow::Result<Vec<String>> {
        let content = fs::read_to_string(path)?;
        let mut keys = Vec::new();
        for raw in content.lines() {
            let line = raw.trim();
            if line.is_empty() || line.starts_with('#') { continue; }
            if let Some((k, _)) = line.split_once('=') {
                let key = k.trim();
                if !key.is_empty() {
                    keys.push(key.to_string());
                }
            }
        }
        Ok(keys)
    }

  <files>ferro-cli/src/deploy/rewrite_ferro_version.rs, ferro-cli/src/deploy/mod.rs</files>
  <behavior>
    - Given a Cargo.toml with `ferro = { path = "../../framework" }` and ferro_version_override = Some("0.1.87"), writes Cargo.docker.toml containing `ferro = { version = "0.1.87" }` (or equivalent version-only form).
    - Given no override, reads the path dep's actual workspace Cargo.toml `package.version` and uses that.
    - Given no override and unreadable path-dep version, falls back to `version = "*"`.
    - Non-ferro deps are left untouched.
    - Multiple ferro* deps (ferro, ferro-macros, ferro-events, etc.) all rewritten.
    - Returns the PathBuf of the written file ({project_root}/Cargo.docker.toml).
  </behavior>
  <action>
    Create ferro-cli/src/deploy/rewrite_ferro_version.rs. Use the `toml_edit` crate if already a ferro-cli dep (preserves formatting). If not, use `toml` parse + serialize; formatting change is acceptable.

    Target signature:
    ```rust
    pub fn rewrite_cargo_docker_toml(
        project_root: &Path,
        ferro_version_override: Option<&str>,
    ) -> anyhow::Result<PathBuf>
    ```

    Algorithm:
    1. Read `{project_root}/Cargo.toml` into a toml_edit::Document.
    2. For each `[dependencies]` (and `[dev-dependencies]` and `[build-dependencies]`) entry whose key starts with `ferro` and whose value is an inline table with a `path` key:
       - Determine version string:
         a. If `ferro_version_override` is Some, use it.
         b. Else: read `{path_dep}/Cargo.toml` → `package.version`.
         c. Else (any error): use `"*"`.
       - Replace the entry value with `{ version = "X" }` (inline table with only `version`).
    3. Serialize to `{project_root}/Cargo.docker.toml`.
    4. Return that path.

    Register `pub mod rewrite_ferro_version;` in ferro-cli/src/deploy/mod.rs.

    Add tests using tempfile with a fake project + fake ferro workspace sibling for cases a, b, and c above, plus a "multiple ferro* deps" case and a "non-ferro deps unchanged" case.

    Per SCOPE §2 (rewriter). ~30 LOC of logic + ~50 LOC of tests.
  </action>
