```rust
// ferro-cli/src/deploy/env_example.rs
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct EnvEntry {
    pub key: String,
    pub value: String, // raw value, no unquoting
}

/// Parse a .env-style string. Rules:
/// - Skip blank lines and lines whose first non-whitespace char is `#`.
/// - Split each remaining line on the FIRST `=`. Left = key (trimmed),
///   right = value (trimmed, kept verbatim including any surrounding quotes).
/// - Lines without `=` are silently skipped.
/// - Preserves original order; duplicates kept as-is.
pub fn parse_env_example(content: &str) -> Vec<EnvEntry>;
```

```rust
// ferro-cli/src/deploy/classify.rs
/// A key is a SECRET when, uppercased:
///   - ends with "_KEY", "_SECRET", "_PASSWORD", or "_TOKEN", OR
///   - equals "DATABASE_URL".
pub fn is_secret(key: &str) -> bool;
```

```rust
// ferro-cli/src/deploy/ferro_deps.rs
/// Render the full contents of scripts/rewrite-ferro-deps.sh for a
/// project rooted at the directory containing `cargo_toml`. The script,
/// when run from the project root at Docker build time, edits Cargo.toml
/// in place, replacing every `ferro*` dep that currently uses
/// `path = "..."` with a git dep pinned to `ferro_ref`.
///
/// The generated script:
///   - starts with `#!/usr/bin/env bash` and `set -euo pipefail`
///   - embeds a header comment "# Generated by ferro docker:init — do not edit"
///   - embeds the ferro_ref literal in a `FERRO_REF="..."` variable
///   - uses `sed -i` with per-crate patterns to rewrite every ferro* path dep
///     that was discovered in cargo_toml at generation time
///
/// cargo_toml is read only to enumerate which ferro* crate names to target.
pub fn render_rewrite_script(cargo_toml: &std::path::Path, ferro_ref: &str)
    -> std::io::Result<String>;
```
