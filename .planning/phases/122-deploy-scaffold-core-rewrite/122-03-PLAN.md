// ferro-cli/src/templates/docker.rs

use crate::project::{BinEntry, ProjectDirs};

pub struct DockerfileContext<'a> {
    pub package_name: &'a str,          // used for image label only; binaries come from bins
    pub bins: &'a [BinEntry],           // every [[bin]] entry, empty -> synthesize from package_name
    pub dirs: ProjectDirs,              // which optional dirs to COPY
    pub runtime_deps: &'a [String],     // e.g. ["chromium", "fonts-liberation"]
    pub rust_base_image: &'a str,       // e.g. "rust:1.88-slim-bookworm"
    pub workspace_members: &'a [String],// relative dirs to COPY in planner+builder stages
    pub ferro_ref: &'a str,             // persisted into generated Dockerfile comment
}

pub fn render_dockerfile(ctx: &DockerfileContext) -> String;
pub fn render_dockerignore(extra_entries: &[&str]) -> String;  // unchanged signature; impl in plan 06
```

Dockerfile template placeholders (all BRACE-delimited `{name}`):
- `{frontend_stage}` — entire frontend stage or empty string
- `{rust_base_image}` — e.g. `rust:1.88-slim-bookworm`
- `{workspace_copy_planner}` — `COPY` lines for workspace members in planner stage (joined with `\n`)
- `{workspace_copy_builder}` — same, builder stage
- `{ferro_rewrite_planner}` — `RUN bash scripts/rewrite-ferro-deps.sh` or empty
- `{ferro_rewrite_builder}` — same, builder stage
- `{cargo_build_bins}` — one `--bin <name>` flag joined by space, passed to `cargo build --release`
- `{runtime_apt_block}` — the marked runtime deps apt block or empty string
- `{runtime_bin_copies}` — one `COPY --from=backend-builder --chown=appuser:appuser /app/target/release/<bin> /usr/local/bin/<bin>` per BinEntry
- `{runtime_optional_copies}` — conditional COPY lines for themes, lang, public, migrations
- `{entrypoint_bin}` — server bin name (first matching package_name, else bins[0])
- `{ferro_ref}` — e.g. `main` (comment only)
</interfaces>

    Helpers (private):
    - `fn frontend_stage_block(has_frontend: bool) -> String` — returns the full `FROM node:22-slim AS frontend-builder` block (mirroring the current Dockerfile.tpl lines 1-14) or `String::new()`.

      ```
      # >>> ferro:runtime-deps (regenerated by ferro docker:init --runtime-deps=...)
      RUN apt-get update && apt-get install -y --no-install-recommends \
          <dep1> <dep2> ... \
          && rm -rf /var/lib/apt/lists/*
      # <<< ferro:runtime-deps
