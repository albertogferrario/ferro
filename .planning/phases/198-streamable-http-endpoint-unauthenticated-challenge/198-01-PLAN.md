    ```rust
    use crate::config::McpServerConfig;
    use crate::{dispatch, McpContext, render_exposed_tools};
    use ferro_projections::ServiceDef;
    use serde_json::{json, Value};

    /// initialize — minimal spec-compliant result (D-03).
    /// protocolVersion literal is "2025-03-26" (rmcp 0.12 ProtocolVersion::LATEST).
    pub async fn handle_initialize(_params: Value, config: &McpServerConfig) -> Value {
        json!({
            "result": {
                "protocolVersion": "2025-03-26",
                "capabilities": { "tools": {} },
                "serverInfo": { "name": config.app_name, "version": config.version }
            }
        })
    }

    /// tools/list — render the mcp_exposed projections to MCP tools.
    pub async fn handle_tools_list(services: &[ServiceDef], _config: &McpServerConfig) -> Value {
        match render_exposed_tools(services, &McpContext) {
            Ok(tools) => json!({ "result": { "tools": tools } }),
            Err(e) => json!({ "error": { "code": -32603, "message": e.to_string() } }),
        }
    }

    /// tools/call — strip "list_" prefix (renderer names tools "list_{name}"),
    /// resolve the exposed ServiceDef, run dispatch. Pagination + filters come from `arguments`.
    pub async fn handle_tools_call(
        call_params: Value,
        services: &[ServiceDef],
        db: &sea_orm::DatabaseConnection,
    ) -> Value {
        let tool_name = call_params["name"].as_str().unwrap_or("");
        let service_name = tool_name.strip_prefix("list_").unwrap_or(tool_name);
        let service = match services.iter().find(|s| s.name == service_name && s.mcp_exposed) {
            Some(s) => s,
            None => {
                return json!({ "error": { "code": -32601, "message": "Method not found" } })
            }
        };
        let args = call_params.get("arguments").cloned().unwrap_or_else(|| json!({}));
        let limit = args.get("limit").and_then(Value::as_u64).unwrap_or(25);
        let offset = args.get("offset").and_then(Value::as_u64).unwrap_or(0);
        // Filters = arguments minus the pagination keys. dispatch enforces the allowlist
        // (is_filter_field) + MAX_LIMIT clamp itself (Phase 197 WR-01/WR-02) — do not re-guard.
        let mut filters = args.clone();
        if let Some(obj) = filters.as_object_mut() {
            obj.remove("limit");
            obj.remove("offset");
        }
        match dispatch(service, filters, limit, offset, db).await {
            Ok(result) => json!({
                "result": {
                    "content": result.rows,
                    "total": result.total,
                    "limit": result.limit,
                    "offset": result.offset
                }
            }),
            Err(e) => json!({ "error": { "code": -32602, "message": e.to_string() } }),
        }
    }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-198-01 | Spoofing (V2 Authn) | `auth::extract_bearer` | mitigate | Seam ALWAYS returns `Unauthenticated` in Phase 198; no branch constructs `Authenticated`. Unit test `any_bearer_is_still_unauthenticated_in_phase_198` asserts this. Phase 199 fills the valid-token path. |
| T-198-02 | Tampering / Information Disclosure (V5 Input Validation) | `jsonrpc::handle_tools_call` → `dispatch` | mitigate | Filter keys validated by Phase 197 `is_filter_field` allowlist inside `dispatch`; `limit` clamped to `MAX_LIMIT`. jsonrpc.rs does NOT re-implement or weaken these guards — acceptance criterion greps that `is_filter_field`/`MAX_LIMIT` are absent from jsonrpc.rs. |
| T-198-03 | Denial of Service | `tools/call` unbounded result set | mitigate | `dispatch` clamps `limit` to `MAX_LIMIT`=100 regardless of caller-supplied value; pagination keys stripped from filters before dispatch. |
| T-198-04 | Elevation / unknown-method probing | `handle_tools_call` tool resolution | mitigate | Unknown / non-`mcp_exposed` tool name → JSON-RPC `-32601` (method not found); only `mcp_exposed == true` services are reachable. |
| T-198-05 | DNS rebinding (Origin header) | live HTTP surface | accept (deferred) | Phase 198 is test-only (no live server); Origin validation deferred to Phase 199. The live handler in Plan 02 carries `// TODO(phase-199): validate Origin header`. |
