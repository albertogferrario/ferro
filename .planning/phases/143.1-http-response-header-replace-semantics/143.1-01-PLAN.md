        /// Set a response header, replacing any existing header with the same name.
        ///
        /// The name match is case-insensitive (ASCII). Use [`append_header`](Self::append_header)
        /// for legitimately multi-value headers such as `Set-Cookie`, `Vary`, or `Link`.
        pub fn header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
            let name = name.into();
            self.headers.retain(|(n, _)| !n.eq_ignore_ascii_case(&name));
            self.headers.push((name, value.into()));
            self
        }

        /// Append a response header without removing any existing entry with the same name.
        ///
        /// Intended for headers that legitimately carry multiple values on separate lines,
        /// such as `Set-Cookie` (RFC 6265 §4.1), `Vary`, and `Link`. For single-value
        /// headers like `Content-Type` or `Location`, use [`header`](Self::header) instead.
        pub fn append_header(mut self, name: impl Into<String>, value: impl Into<String>) -> Self {
            self.headers.push((name.into(), value.into()));
            self
        }

        /// Get the response headers as a borrowed slice.
        ///
        /// Returns all header entries in insertion order. Multi-value headers
        /// (e.g. `Set-Cookie`) appear as multiple entries with the same name.
        pub fn headers(&self) -> &[(String, String)] {
            &self.headers
        }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| App handler → `HttpResponse::header(name, value)` | Name and value are `impl Into<String>`. In practice the name is a framework-internal constant (`"Content-Type"`, `"Location"`, `"X-Inertia"`, `"Set-Cookie"`); callers that pass user-controlled names or values already have a header-injection risk that predates this phase. |
| App handler → `HttpResponse::append_header(name, value)` | New pub API. Same threat surface as `header()`, but `retain` is not involved, so `append_header` contributes no new attack path beyond `header`. |
| `HttpResponse::into_hyper()` → `hyper::Response::builder().header(name, value)` | Final sanitization boundary. `hyper` rejects header names and values containing CR/LF/NUL; an invalid value produces a `builder.body().unwrap()` panic (existing behaviour, unchanged by this phase). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-143.1-01 | Tampering (header injection via user-controlled name) | `HttpResponse::header(name, value)` when `name` contains CR/LF | accept | Out of scope for this phase. The framework already relies on `hyper`'s header builder to reject invalid names at `into_hyper()`; no caller in the workspace passes user input as a header name. Behaviour is unchanged by this phase — `retain` performs an `eq_ignore_ascii_case` comparison on the stored Vec, which does not execute or emit the string. No new attack surface. |
| T-143.1-02 | Tampering (CRLF injection via user-controlled value) | `HttpResponse::header(name, value)` when `value` contains `\r\n` | accept | Same as T-143.1-01. `hyper` sanitizes at serialization. No caller in the workspace passes raw user input as a header value; `HttpResponse::download` already strips control characters from filenames at construction (response.rs:63-66). Unchanged by this phase. |
| T-143.1-03 | Information Disclosure (retain on user-supplied name reveals a secret via timing) | `self.headers.retain(|(n, _)| !n.eq_ignore_ascii_case(&name))` | accept | `eq_ignore_ascii_case` is a non-constant-time comparison, but it operates on header names (e.g. `"Content-Type"`, `"Set-Cookie"`) that are public framework constants. No secret material is ever stored as a header name. Timing the comparison reveals nothing confidential. |
| T-143.1-04 | Denial of Service (pathological Vec growth via repeated append_header) | `HttpResponse::append_header` called in a loop with user-controlled input | accept | `append_header` is pub but only invoked internally by `cookie()` (at most once per `Set-Cookie`) and by application code the developer writes.
