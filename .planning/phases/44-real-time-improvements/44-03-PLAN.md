**framework/src/broadcast/auth.rs:**

The `broadcasting_auth` handler:
1. Checks the user is authenticated via `Auth::id()`
2. Extracts `channel_name` and `socket_id` from the POST body
3. Looks up the Broadcaster from the App container
4. Calls the registered ChannelAuthorizer with the user_id as the auth_token
5. Returns 200 with auth confirmation on success, 403 on failure

```rust
use crate::auth::Auth;
use crate::container::App;
use crate::error::FrameworkError;
use crate::http::{HttpResponse, Request, Response};
use ferro_broadcast::{AuthData, Broadcaster};

/// Broadcasting channel authorization handler.
///
/// Register this as a POST route in your routes file. It MUST run through
/// session middleware so that `Auth::id()` works.
///
/// # Example
///
/// ```rust,ignore
/// use ferro::broadcasting_auth;
///
/// Route::post("/broadcasting/auth", broadcasting_auth)
///     .middleware(SessionAuthMiddleware);
/// ```
///
/// # Protocol
///
/// Clients send POST with `channel_name` and `socket_id` form fields.
/// Returns 200 with `{"auth": "ok"}` if authorized, 403 otherwise.
///
/// For presence channels, include `channel_data` in the response with
/// user info for member tracking.
#[ferro_macros::handler]
pub async fn broadcasting_auth(req: Request) -> Response {
    // Verify authenticated
    let user_id = Auth::id()
        .ok_or_else(|| FrameworkError::domain("Unauthenticated".to_string(), 401))?;

    // Parse form body
    let form = req.input().await?;
    let channel_name: String = form.get("channel_name")
        .ok_or_else(|| FrameworkError::domain("Missing channel_name".to_string(), 422))?;
    let socket_id: String = form.get("socket_id")
        .ok_or_else(|| FrameworkError::domain("Missing socket_id".to_string(), 422))?;

    // Get broadcaster
    let broadcaster = App::get::<Broadcaster>()
        .ok_or_else(|| FrameworkError::internal("Broadcasting not configured".to_string()))?;

    // Build auth data with user_id as the auth token
    let auth_data = AuthData {
        socket_id: socket_id.clone(),
        channel: channel_name.clone(),
        auth_token: Some(user_id.to_string()),
    };

    // Check authorization via registered authorizer
    // If no authorizer is set, deny by default for private/presence channels
    let channel_type = ferro_broadcast::ChannelType::from_name(&channel_name);
    if channel_type.requires_auth() {
        let authorized = broadcaster.check_auth(&auth_data).await;
        if !authorized {
            return Err(HttpResponse::json(serde_json::json!({
                "error": "Forbidden"
            })).status(403));
        }
    }

    // Build response
    let mut response = serde_json::json!({
        "auth": "ok",
        "socket_id": socket_id,
        "channel": channel_name,
    });

    // For presence channels, include user info
    if channel_type == ferro_broadcast::ChannelType::Presence {
        response["channel_data"] = serde_json::json!({
            "user_id": user_id.to_string(),
        });
    }

    Ok(HttpResponse::json(response))
}
```

Add the `check_auth` public method to Broadcaster (as described in Task 1). This method checks authorization without subscribing — it's a read-only auth check for the HTTP auth endpoint.

```rust
/// Check if a client would be authorized for a channel.
///
/// Returns true if:
/// - Channel is public (no auth needed)
/// - Channel is private/presence AND authorizer returns true
/// - Channel is private/presence AND no authorizer registered → returns false
///
/// Used by the broadcasting auth HTTP endpoint.
pub async fn check_auth(&self, auth_data: &AuthData) -> bool {
    let channel_type = ChannelType::from_name(&auth_data.channel);
    if !channel_type.requires_auth() {
        return true;
    }
    if let Some(authorizer) = &self.inner.authorizer {
        authorizer.authorize(auth_data).await
    } else {
        false
    }
}
```
