**Key function: `handle_ws_upgrade`**

Takes the raw hyper::Request, performs the WebSocket upgrade via hyper_tungstenite, spawns the connection handler task, and returns the HTTP upgrade response.

```rust
use crate::container::App;
use bytes::Bytes;
use ferro_broadcast::{Broadcaster, ClientMessage, ServerMessage};
use futures_util::{SinkExt, StreamExt};
use http_body_util::Full;
use std::sync::Arc;
use tokio::sync::mpsc;
use tokio::time::Instant;
use uuid::Uuid;

pub fn handle_ws_upgrade(
    mut req: hyper::Request<hyper::body::Incoming>,
) -> hyper::Response<Full<Bytes>> {
    // Get broadcaster from App container
    let broadcaster = match App::get::<Broadcaster>() {
        Some(b) => b,
        None => {
            return hyper::Response::builder()
                .status(503)
                .body(Full::new(Bytes::from("Broadcasting not configured")))
                .unwrap();
        }
    };

    let (response, ws_future) = match hyper_tungstenite::upgrade(&mut req, None) {
        Ok(pair) => pair,
        Err(e) => {
            eprintln!("WebSocket upgrade failed: {}", e);
            return hyper::Response::builder()
                .status(400)
                .body(Full::new(Bytes::from("WebSocket upgrade failed")))
                .unwrap();
        }
    };

    tokio::spawn(async move {
        match ws_future.await {
            Ok(ws_stream) => {
                handle_connection(ws_stream, broadcaster).await;
            }
            Err(e) => {
                eprintln!("WebSocket connection failed: {}", e);
            }
        }
    });

    response
}
```

**Key function: `handle_connection`**

Runs the message loop with heartbeat/timeout using `tokio::select!`.

```rust
async fn handle_connection(
    ws_stream: hyper_tungstenite::WebSocketStream<hyper_tungstenite::hyper::upgrade::Upgraded>,
    broadcaster: Broadcaster,
) {
    let socket_id = Uuid::new_v4().to_string();
    let (mut ws_write, mut ws_read) = ws_stream.split();
    let (tx, mut rx) = mpsc::channel::<ServerMessage>(32);

    // Register client
    broadcaster.add_client(socket_id.clone(), tx);

    // Send connection_established
    let connected = ServerMessage::Connected { socket_id: socket_id.clone() };
    if let Ok(msg) = connected.to_ws_message() {
        let _ = ws_write.send(msg).await;
    }

    let config = broadcaster.config().clone();
    let mut heartbeat_interval = tokio::time::interval(config.heartbeat_interval);
    let mut last_activity = Instant::now();

    loop {
        tokio::select! {
            // Incoming WebSocket message
            frame = ws_read.next() => {
                match frame {
                    Some(Ok(msg)) => {
                        last_activity = Instant::now();
                        match msg {
                            tokio_tungstenite::tungstenite::Message::Text(text) => {
                                handle_client_message(&text, &socket_id, &broadcaster, &mut ws_write).await;
                            }
                            tokio_tungstenite::tungstenite::Message::Close(_) => break,
                            tokio_tungstenite::tungstenite::Message::Pong(_) => {
                                // Pong received, activity already updated
                            }
                            _ => {} // tungstenite handles Ping automatically with Pong
                        }
                    }
                    Some(Err(_)) => break, // Connection error
                    None => break,         // Stream ended
                }
            }
            // Server message to forward to client
            server_msg = rx.recv() => {
                match server_msg {
                    Some(msg) => {
                        if let Ok(ws_msg) = msg.to_ws_message() {
                            if ws_write.send(ws_msg).await.is_err() {
                                break; // Write failed, connection dead
                            }
                        }
                    }
                    None => break, // Channel closed
                }
            }
            // Heartbeat tick
            _ = heartbeat_interval.tick() => {
                if last_activity.elapsed() > config.client_timeout {
                    break; // Client timed out
                }
                // Send WS-level Ping (tungstenite auto-responds with Pong on the other end)
                use tokio_tungstenite::tungstenite::Message as WsMsg;
                if ws_write.send(WsMsg::Ping(vec![].into())).await.is_err() {
                    break; // Write failed
                }
            }
        }
    }

    // Clean up: send Close frame, remove from broadcaster
    use tokio_tungstenite::tungstenite::Message as WsMsg;
    let _ = ws_write.send(WsMsg::Close(None)).await;
    broadcaster.remove_client(&socket_id);
}
```

**Key function: `handle_client_message`**

Parses ClientMessage from JSON text and dispatches to broadcaster.

```rust
async fn handle_client_message(
    text: &str,
    socket_id: &str,
    broadcaster: &Broadcaster,
    ws_write: &mut impl SinkExt<tokio_tungstenite::tungstenite::Message, Error = tokio_tungstenite::tungstenite::Error> + Unpin,
) {
    let client_msg = match ClientMessage::from_ws_text(text) {
        Ok(msg) => msg,
        Err(_) => {
            // Send error back
            let err = ServerMessage::Error { message: "Invalid message format".into() };
            if let Ok(ws_msg) = err.to_ws_message() {
                let _ = ws_write.send(ws_msg).await;
            }
            return;
        }
    };

    match client_msg {
        ClientMessage::Subscribe { channel, auth } => {
            match broadcaster.subscribe(socket_id, &channel, auth.as_deref(), None).await {
                Ok(()) => {
                    let msg = ServerMessage::Subscribed { channel };
                    if let Ok(ws_msg) = msg.to_ws_message() {
                        let _ = ws_write.send(ws_msg).await;
                    }
                }
                Err(e) => {
                    let msg = ServerMessage::SubscriptionError {
                        channel,
                        error: e.to_string(),
                    };
                    if let Ok(ws_msg) = msg.to_ws_message() {
                        let _ = ws_write.send(ws_msg).await;
                    }
                }
            }
        }
        ClientMessage::Unsubscribe { channel } => {
            broadcaster.unsubscribe(socket_id, &channel).await;
            let msg = ServerMessage::Unsubscribed { channel };
            if let Ok(ws_msg) = msg.to_ws_message() {
                let _ = ws_write.send(ws_msg).await;
            }
        }
        ClientMessage::Whisper { channel, event, data } => {
            if let Err(e) = broadcaster.whisper(socket_id, &channel, &event, data).await {
                let msg = ServerMessage::Error { message: e.to_string() };
                if let Ok(ws_msg) = msg.to_ws_message() {
                    let _ = ws_write.send(ws_msg).await;
                }
            }
        }
        ClientMessage::Ping => {
            let msg = ServerMessage::Pong;
            if let Ok(ws_msg) = msg.to_ws_message() {
                let _ = ws_write.send(ws_msg).await;
            }
        }
    }
}
```

**Important implementation details:**
- Use `App::get::<Broadcaster>()` to retrieve the broadcaster from the DI container. Users register it with `App::singleton(broadcaster)` in their bootstrap.
- The handler returns a `hyper::Response<Full<Bytes>>` directly, NOT a framework HttpResponse — this is a raw hyper response for the upgrade handshake.
- Heartbeat uses WS-level Ping frames (tungstenite auto-responds), NOT application-level JSON ping.
- Track `last_activity` on ANY incoming frame (text, pong), not just pong.
- Always send Close frame before calling remove_client for clean shutdown.

**In framework/src/lib.rs:** Add `pub mod websocket;` and re-export nothing publicly — the module is used internally by server.rs only. Actually, do NOT add pub mod — add `mod websocket;` in server.rs as a private module, or `pub(crate) mod websocket;` in lib.rs. Prefer keeping it alongside server.rs: add `mod websocket;` at the top of lib.rs (or as a sibling module). The simplest approach: add `mod websocket;` in lib.rs (no pub), and in server.rs add `use crate::websocket::handle_ws_upgrade;`.
