// Internal parse-wire struct wrapping elements in a duplicate-rejecting map.
#[derive(Deserialize)]
struct SpecWire {
    #[serde(rename = "$schema", default = "default_schema")]
    schema: String,
    root: String,
    elements: ElementsMap,
    #[serde(default)]
    title: Option<String>,
    #[serde(default)]
    layout: Option<String>,
    #[serde(default)]
    data: Value,
}

fn default_schema() -> String {
    SCHEMA_VERSION.to_string()
}

// Wrapper that fails on duplicate keys (D-12, RESEARCH.md Pitfall 2 Approach A).
struct ElementsMap(HashMap<String, Element>);

impl<'de> DeserializeTrait<'de> for ElementsMap {
    fn deserialize<D: Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
        struct V;
        impl<'de> Visitor<'de> for V {
            type Value = ElementsMap;
            fn expecting(&self, f: &mut fmt::Formatter) -> fmt::Result {
                f.write_str("a JSON object with unique element IDs")
            }
            fn visit_map<M: MapAccess<'de>>(self, mut m: M) -> Result<ElementsMap, M::Error> {
                let mut map: HashMap<String, Element> = HashMap::new();
                while let Some(k) = m.next_key::<String>()? {
                    if map.contains_key(&k) {
                        return Err(serde::de::Error::custom(
                            format!("__FERRO_DUPLICATE_ID__{k}")
                        ));
                    }
                    let v: Element = m.next_value()?;
                    map.insert(k, v);
                }
                Ok(ElementsMap(map))
            }
        }
        d.deserialize_map(V)
    }
}

impl Spec {
    pub fn from_json(json: &str) -> Result<Spec, SpecError> {
        let raw: SpecWire = match serde_json::from_str(json) {
            Ok(r) => r,
            Err(e) => {
                // Intercept the custom duplicate-ID sentinel and convert to DuplicateId variant.
                let msg = e.to_string();
                if let Some(idx) = msg.find("__FERRO_DUPLICATE_ID__") {
                    let after = &msg[idx + "__FERRO_DUPLICATE_ID__".len()..];
                    let id: String = after.chars().take_while(|c| !c.is_whitespace() && *c != '"' && *c != '\'').collect();
                    return Err(SpecError::DuplicateId(id));
                }
                return Err(SpecError::Json(e));
            }
        };
        let spec = Spec {
            schema: raw.schema,
            root: raw.root,
            elements: raw.elements.0,
            title: raw.title,
            layout: raw.layout,
            data: raw.data,
        };
        validate_structure(&spec)?;
        Ok(spec)
    }
}

fn validate_structure(spec: &Spec) -> Result<(), SpecError> {
    validate_ids(&spec.elements)?;
    if !spec.elements.contains_key(&spec.root) {
        return Err(SpecError::RootMissing(spec.root.clone()));
    }
    validate_no_dangling(&spec.elements)?;
    detect_cycle(&spec.elements, &spec.root)?;
    check_depth(&spec.elements, &spec.root)?;
    Ok(())
}

fn is_valid_id(s: &str) -> bool {
    if s.is_empty() || s.len() > 128 { return false; }
    let bytes = s.as_bytes();
    let first = bytes[0];
    let first_ok = first.is_ascii_alphabetic() || first == b'_';
    if !first_ok { return false; }
    bytes[1..].iter().all(|&b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
}

fn validate_ids(elements: &HashMap<String, Element>) -> Result<(), SpecError> {
    for (id, el) in elements {
        if !is_valid_id(id) { return Err(SpecError::InvalidId(id.clone())); }
        for child in &el.children {
            if !is_valid_id(child) { return Err(SpecError::InvalidId(child.clone())); }
        }
    }
    Ok(())
}

fn validate_no_dangling(elements: &HashMap<String, Element>) -> Result<(), SpecError> {
    for (id, el) in elements {
        for child in &el.children {
            if !elements.contains_key(child) {
                return Err(SpecError::DanglingChild { element: id.clone(), child: child.clone() });
            }
        }
    }
    Ok(())
}

fn detect_cycle(elements: &HashMap<String, Element>, root: &str) -> Result<(), SpecError> {
    let mut visited: HashSet<String> = HashSet::new();
    let mut on_stack: Vec<String> = Vec::new();
    dfs(root, elements, &mut visited, &mut on_stack)
}

fn dfs(node: &str, elements: &HashMap<String, Element>, visited: &mut HashSet<String>, on_stack: &mut Vec<String>) -> Result<(), SpecError> {
    if on_stack.iter().any(|n| n == node) {
        let start = on_stack.iter().position(|n| n == node).unwrap();
        let mut path: Vec<String> = on_stack[start..].to_vec();
        path.push(node.to_string());
        return Err(SpecError::Cycle { path });
    }
    if visited.contains(node) { return Ok(()); }
    on_stack.push(node.to_string());
    if let Some(el) = elements.get(node) {
        for child in &el.children {
            dfs(child, elements, visited, on_stack)?;
        }
    }
    on_stack.pop();
    visited.insert(node.to_string());
    Ok(())
}

fn check_depth(elements: &HashMap<String, Element>, root: &str) -> Result<(), SpecError> {
    let mut path: Vec<String> = Vec::new();
    walk(root, elements, 1, &mut path)
}

fn walk(node: &str, elements: &HashMap<String, Element>, depth: usize, path: &mut Vec<String>) -> Result<(), SpecError> {
    path.push(node.to_string());
    if depth > MAX_NESTING_DEPTH {
        return Err(SpecError::DepthExceeded { max: MAX_NESTING_DEPTH, found: depth, path: path.clone() });
    }
    if let Some(el) = elements.get(node) {
        for child in &el.children {
            walk(child, elements, depth + 1, path)?;
        }
    }
    path.pop();
    Ok(())
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller → `Spec::from_json(&str)` | Untrusted JSON string (will be called from the page loader in Phase 119). MUST NOT panic on any input. MUST reject deeply-nested, cyclic, or malformed specs with a typed `SpecError`. |

## Threats

| ID | Severity | Asset | Attack | Mitigation |
|----|----------|-------|--------|------------|
| T-115-01 | med | `Spec::from_json` caller | DoS via deeply nested spec — exponential descent | `check_depth` rejects any path > `MAX_NESTING_DEPTH = 3`; recursion bounded at 4 frames |
| T-115-02 | med | `Spec::from_json` caller | DoS via cyclic element graph — infinite recursion | `detect_cycle` runs BEFORE `check_depth` with DFS + on-stack set; any back-edge returns `SpecError::Cycle` |
| T-115-03 | low | Author | Silent data loss via duplicate IDs in raw JSON (serde's default overwrites) | Custom `ElementsMap` visitor detects duplicates at parse time and emits `SpecError::DuplicateId(id)` |
| T-115-04 | high | `Spec::from_json` caller | Panic on malformed input → process abort | Structural validation returns `Result<Spec, SpecError>`. No `unwrap`/`expect` on any user-influenced value in `from_json` / `validate_structure` / `dfs` / `walk`. `serde_json::from_str` errors wrap into `SpecError::Json(#[from])` |
| T-115-05 | low | Memory | Very large element map (tens of thousands of elements) consumes unbounded memory | Accepted. A max-element sanity cap is deferred — see note below. `check_depth` caps recursion but not total element count. |

**T-115-05 disposition rationale:** The 3-level depth cap caps the *useful* element count (branching_factor^3). A malicious caller could still construct a 100k-element flat map all siblings of root; `check_depth` completes in O(n) and does not allocate beyond the traversal path. No immediate DoS vector at this layer; rate-limiting is the appropriate mitigation, applied in Phase 119's loader — not here. Flagging explicitly so Phase 119 planning cannot silently inherit the assumption.

Security enforcement: all threats with `mitigate` disposition have concrete implementation references in the task actions above (file + function). T-115-05 is `accept` with explicit rationale.
