    ```rust
    use std::collections::HashMap;

    use sea_orm::DatabaseConnection;
    use serde::Serialize;
    use serde_json::Value;

    use ferro_projections::{schema_contract, SchemaContract, ServiceDef};

    use crate::http::Request;
    use crate::inertia::context::Inertia;
    use crate::permitted_actions::permitted_actions;
    use crate::projection_read::{dispatch, DispatchResult};
    use crate::Response;

    /// Query parameters for a projection-driven Inertia page.
    /// Default: filters = {}, limit = 25, offset = 0 (matches the MCP data defaults).
    #[derive(Debug, Clone)]
    pub struct ProjectionQuery {
        pub filters: Value,
        pub limit: u64,
        pub offset: u64,
    }

    impl Default for ProjectionQuery {
        fn default() -> Self {
            Self { filters: Value::Object(Default::default()), limit: 25, offset: 0 }
        }
    }

    // consuming builder methods (mut self -> Self) per crate convention:
    impl ProjectionQuery {
        pub fn filters(mut self, f: Value) -> Self { self.filters = f; self }
        pub fn limit(mut self, n: u64) -> Self { self.limit = n; self }
        pub fn offset(mut self, n: u64) -> Self { self.offset = n; self }
    }

    #[derive(Debug, Serialize)]
    struct ProjectionProps {
        schema: SchemaContract,
        data: Vec<Value>,
        permitted_actions: Vec<String>,
        total: u64,
        limit: u64,
        offset: u64,
    }

    impl Inertia {
        /// Render an Inertia response from a projection declaration.
        ///
        /// Assembles `{ schema, data, permitted_actions, total, limit, offset }` from `service`
        /// and `evaluated_guards`, then delegates to `Inertia::render`.
        ///
        /// `permitted_actions` here is a per-REQUEST visibility list (not per-record). Actual
        /// per-record guard enforcement happens at `dispatch_write` time. This helper adds READ
        /// derivation only — it changes no write path.
        pub async fn from_projection(
            req: &Request,
            component: &str,
            service: &ServiceDef,
            query: ProjectionQuery,
            db: &DatabaseConnection,
            tenant_id: Option<i64>,
            evaluated_guards: &HashMap<String, bool>,
        ) -> Response {
            let schema = schema_contract(service);
            let actions = permitted_actions(service, evaluated_guards);

            let result: DispatchResult = match dispatch(
                service, query.filters, query.limit, query.offset, db, tenant_id,
            ).await {
                Ok(r) => r,
                Err(e) => {
                    return Inertia::render(req, component, serde_json::json!({ "error": e.to_string() }));
                }
            };

            let props = ProjectionProps {
                schema,
                data: result.rows,
                permitted_actions: actions,
                total: result.total,
                limit: result.limit,
                offset: result.offset,
            };
            Inertia::render(req, component, props)
        }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → Inertia read (from_projection) | Request-supplied `ProjectionQuery` (filters/limit/offset) and the authenticated `tenant_id` cross into the tenant-scoped data query. |
| client → Inertia write (POST /{service}/{action}) | Form input crosses into `dispatch_write(channel="web")` where guards are re-evaluated server-side. |
| props → frontend | `permitted_actions` in props is advisory display data, NOT an authorization grant. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-263-10 | Information disclosure (cross-tenant) | `from_projection` returns rows/actions for another tenant | mitigate | Data load goes through `framework::projection_read::dispatch` with the authenticated `tenant_id`; cross-tenant ids are not found (tenant-scoping tests in Plan 03/05). `from_projection` never bypasses the tenant predicate. |
| T-263-11 | Elevation of privilege (authz bypass) | Frontend treats `permitted_actions` as permission and the Inertia surface becomes a write bypass | mitigate | Writes go ONLY through the existing `POST /{service}/{action}` → `dispatch_write(channel="web")`, which re-evaluates guards via the live `GuardEvaluatorFn`. `permitted_actions` is documented advisory-only. No new write path is added (Task 2 asserts exactly one `dispatch_write` call site). |
| T-263-12 | Elevation of privilege (dependency cycle → wrong placement) | Placing `from_projection` in `ferro-inertia` forces a `ferro-inertia → framework` cycle, which could tempt a bypass or a broken build | mitigate | Task 0 self-checks the cycle direction and locks the reconciled framework-side placement; acceptance criteria assert `ferro-inertia` gains no framework dep and imports no `ferro-mcp-server`. |
| T-263-13 | Denial of service | Unbounded `limit` in `ProjectionQuery` | mitigate | `dispatch` enforces `MAX_LIMIT=100` (Plan 03) regardless of the `ProjectionQuery.limit` value; default is 25. |

The Inertia surface adds READ derivation only; all write authorization stays in `dispatch_write`.
