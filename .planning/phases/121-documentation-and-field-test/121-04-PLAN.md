Custom layout registration (Rust, app bootstrap):
```rust
use ferro_json_ui::{register_layout, Layout, LayoutContext};

struct MyLayout;
impl Layout for MyLayout {
    fn render(&self, ctx: &LayoutContext) -> String {
        format!("<html><body>{}</body></html>", ctx.content)
    }
}
// In bootstrap:
register_layout("my-layout", MyLayout);
```

Custom plugin registration (Rust, app bootstrap):
```rust
use ferro_json_ui::{register_plugin, JsonUiPlugin, Asset};

struct ChartPlugin;
impl JsonUiPlugin for ChartPlugin {
    fn render(&self, props: &serde_json::Value) -> String {
        format!(r#"<canvas data-chart='{}'></canvas>"#, props)
    }
    fn assets(&self) -> Vec<Asset> { vec![] }
}
register_plugin("Chart", ChartPlugin);
```
