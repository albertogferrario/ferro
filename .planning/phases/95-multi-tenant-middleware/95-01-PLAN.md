tokio::task_local! {
    static SESSION_CONTEXT: Arc<RwLock<Option<SessionData>>>;
}

pub fn session() -> Option<SessionData> {
    SESSION_CONTEXT
        .try_with(|ctx| {
            ctx.try_read().ok().and_then(|guard| guard.clone())
        })
        .ok()
        .flatten()
}
