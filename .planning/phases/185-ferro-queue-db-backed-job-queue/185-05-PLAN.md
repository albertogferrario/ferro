    ```rust
    use std::collections::HashSet;
    use std::sync::{Arc, Mutex};

    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
    async fn two_workers_claim_each_job_exactly_once() {
        // CRITICAL: NamedTempFile, NOT sqlite::memory: (Pitfall 1)
        let db_file = tempfile::NamedTempFile::new().unwrap();
        let db_url = format!("sqlite://{}?mode=rwc", db_file.path().display());

        let conn1 = Database::connect(&db_url).await.unwrap();
        let conn2 = Database::connect(&db_url).await.unwrap();

        // Migrate on conn1 (both see the same file)
        CreateJobsTable.up(&SchemaManager::new(&conn1)).await.unwrap();

        // Enqueue N=20 jobs (all pending, available now)
        const N: usize = 20;
        for _ in 0..N {
            // db::enqueue(&conn1, "default", "TestJob", "{}", 3, None, None, chrono::Utc::now()).await.unwrap();
        }

        // Two concurrent drain tasks, each claiming until the queue is empty
        async fn drain(conn: sea_orm::DatabaseConnection, worker_id: &'static str, out: Arc<Mutex<Vec<i64>>>) {
            loop {
                match ferro_queue::db::claim(&conn, "default", worker_id).await {
                    Ok(Some(row)) => {
                        out.lock().unwrap().push(row.id);
                        // mark done so it cannot be re-observed (delete-on-success)
                        let _ = ferro_queue::db::delete_job(&conn, row.id).await;
                    }
                    Ok(None) => break,
                    Err(e) => panic!("claim error: {e:?}"),
                }
            }
        }
        let c1 = Arc::new(Mutex::new(Vec::new()));
        let c2 = Arc::new(Mutex::new(Vec::new()));
        let (h1, h2) = (
            tokio::spawn(drain(conn1, "w1", c1.clone())),
            tokio::spawn(drain(conn2, "w2", c2.clone())),
        );
        let _ = tokio::join!(h1, h2);

        // Assert exactly-once: union has no duplicates and covers all N
        let mut all: Vec<i64> = c1.lock().unwrap().clone();
        all.extend(c2.lock().unwrap().iter().cloned());
        let unique: HashSet<i64> = all.iter().cloned().collect();
        assert_eq!(unique.len(), all.len(), "a job was claimed more than once");
        assert_eq!(unique.len(), N, "not all jobs were claimed exactly once");
    }
    ```
