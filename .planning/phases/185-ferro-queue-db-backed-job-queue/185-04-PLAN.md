       ```rust
       type RegisterFn = Box<dyn Fn(&mut WorkerLoop) + Send + Sync>;
       static JOB_REGISTRARS: std::sync::Mutex<Vec<RegisterFn>> = std::sync::Mutex::new(Vec::new());

       impl Queue {
           /// Register a job type to be picked up by the auto-started WorkerLoop.
           pub fn register<J>() where J: Job + serde::de::DeserializeOwned + 'static {
               JOB_REGISTRARS.lock().unwrap().push(Box::new(|w: &mut WorkerLoop| w.register::<J>()));
           }
           pub fn has_registered_jobs() -> bool {
               !JOB_REGISTRARS.lock().unwrap().is_empty()
           }
           pub(crate) fn apply_registrars(w: &mut WorkerLoop) {
               for r in JOB_REGISTRARS.lock().unwrap().iter() { r(w); }
           }
       }
       ```
