use ferro::{ derive_intents, DataType, FieldMeaning, JsonUiRenderer, ServiceDef, VisualContext };
use ferro_projections::render::Renderer;

let service = ServiceDef::new("summary_stats")
    .read_only_field("total_revenue_cents", DataType::Integer, FieldMeaning::Money)
    .read_only_field("order_count", DataType::Integer, FieldMeaning::Quantity)
    .read_only_field("average_order_cents", DataType::Integer, FieldMeaning::Money);
let intents = derive_intents(&service);
let spec = JsonUiRenderer.render(&service, &intents, &VisualContext::default())?;
let merged = spec.merge_data(json!({
    "chart_svg": chart_svg,        // RESEARCH §5 Gap 1: no FieldMeaning for a chart — opaque passthrough
    "period_str": period_str,      // RESEARCH §7 Risk 3: period Tabs is controller-side
    "trend_rows": trend_rows,
    ...
}));
JsonUi::render(&merged, &merged.data.clone())
