    ```rust
    //! ferro ci:init — drop .github/workflows/ci.yml (D-13, D-17, D-21).

    use console::style;
    use std::fs;
    use crate::project::{find_project_root, package_name};
    use crate::templates::ci_workflow::{render_ci_workflow, CiWorkflowContext};

    pub fn run(force: bool) {
        let root = match find_project_root(None) {
            Ok(r) => r,
            Err(_) => { eprintln!("{} Cargo.toml not found", style("Error:").red().bold()); std::process::exit(1); }
        };

        let workflows_dir = root.join(".github/workflows");
        let ci_yml = workflows_dir.join("ci.yml");

        if ci_yml.exists() && !force {
            eprintln!("{} {} already exists (use --force)", style("Error:").red().bold(), ci_yml.display());
            std::process::exit(1);
        }

        if let Err(e) = fs::create_dir_all(&workflows_dir) {
            eprintln!("{} Failed to create {}: {}", style("Error:").red().bold(), workflows_dir.display(), e);
            std::process::exit(1);
        }

        let pkg = package_name(&root);
        let content = render_ci_workflow(&CiWorkflowContext { package_name: &pkg });

        if let Err(e) = fs::write(&ci_yml, content) {
            eprintln!("{} Failed to write {}: {}", style("Error:").red().bold(), ci_yml.display(), e);
            std::process::exit(1);
        }

        println!("{} Generated {}", style("✓").green(), ci_yml.display());
    }
    ```
