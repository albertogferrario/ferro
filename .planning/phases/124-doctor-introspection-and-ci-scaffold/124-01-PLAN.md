    use serde::Deserialize;

    pub const IGNORE_PATTERNS_TOML: &str = include_str!("files/root/ignore_patterns.toml");

    #[derive(Debug, Clone, Deserialize)]
    pub struct IgnorePatterns {
        #[serde(rename = "category")]
        pub categories: Vec<Category>,
    }

    pub struct Category {
        pub name: String,
        #[serde(default)]
        pub for_git: bool,
        #[serde(default)]
        pub for_docker: bool,
        #[serde(default)]
        pub patterns: Vec<String>,
        #[serde(default)]
        pub git_patterns: Vec<String>,
        #[serde(default)]
        pub docker_patterns: Vec<String>,
    }

    pub fn parse_ignore_patterns(toml_str: &str) -> Result<IgnorePatterns, String> { ... }
    pub fn load_default() -> IgnorePatterns { ... }   // parses IGNORE_PATTERNS_TOML, panics on malformed (compile-time guaranteed)
    pub fn render_dockerignore(p: &IgnorePatterns) -> String { ... }
    pub fn render_gitignore(p: &IgnorePatterns) -> String { ... }

    Step 2 — Create `ferro-cli/src/commands/ignore_sync.rs` (D-20):

    ```rust
    //! ferro ignore:sync — reconcile .gitignore + .dockerignore in an existing
    //! project against the canonical ignore_patterns.toml (D-18..D-20).

    use console::style;
    use std::fs;
    use std::path::Path;
    use crate::project::find_project_root;
    use crate::templates::ignore_patterns::{load_default, render_dockerignore, render_gitignore};

    pub fn run(dry_run: bool, force: bool) {
        let root = match find_project_root(None) { ... };
        let patterns = load_default();
        let target_docker = render_dockerignore(&patterns);
        let target_git = render_gitignore(&patterns);

        let docker_path = root.join(".dockerignore");
        let git_path = root.join(".gitignore");

        let docker_drift = current_differs(&docker_path, &target_docker);
        let git_drift = current_differs(&git_path, &target_git);

        if !docker_drift && !git_drift {
            println!("{} .gitignore and .dockerignore already in sync", style("✓").green());
            return;
        }

        if dry_run {
            if docker_drift { println!("{} .dockerignore would change", style("DRIFT").yellow()); print_diff(&docker_path, &target_docker); }
            if git_drift    { println!("{} .gitignore would change",    style("DRIFT").yellow()); print_diff(&git_path,    &target_git); }
            std::process::exit(2); // non-zero so CI can detect drift
        }

        // Refuse to overwrite a non-empty existing file unless --force, EXCEPT
        // when only trailing whitespace differs (treat as auto-fixable).
        // Write both files.
    }
    ```

    `current_differs` reads target file (treat NotFound as drift) and compares byte-for-byte after normalizing trailing newline. `print_diff` is a minimal "expected vs actual" line-count summary (no need for a full diff lib).
