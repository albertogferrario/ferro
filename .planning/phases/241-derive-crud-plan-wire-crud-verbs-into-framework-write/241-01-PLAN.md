```rust
/// Carries no behavior: `framework::write::execute_crud_plan` interprets it into
/// parameterized SQL. Values are `serde_json::Value` because this crate has no

pub enum CrudPlan {
    Create {
        table: String,

        /// then server-set fields (created_at, and Status=initial under an SM).

        id_column: String,
        id_value: serde_json::Value,
        /// Set to `now` by the executor (soft-delete, never a physical DELETE).
        soft_delete_column: String,
        tenant_column: Option<TenantColumn>,
    },
}
```

```rust
/// Derives a pure [`CrudPlan`] for `verb` from the service declaration + agent inputs.
/// Mirrors `derive_transition_plan`: pure, side-effect-free, no I/O. Reuses the
/// Phase 239/240 resolver accessors so the column sets are single-sourced with the
/// schema builders. Phase 241 leaves `tenant_column = None` on every variant (D-09).
pub fn derive_crud_plan(
    svc: &crate::ServiceDef,
    verb: CrudVerb,
    inputs: &serde_json::Value,
) -> Result<CrudPlan, crate::Error> {
    let table = svc.resolved_table();
    let has_sm = svc.state_machine.is_some();
    match verb {
        CrudVerb::Create => {
            if !svc.creatable {
                return Err(crate::Error::VerbNotEnabled(format!("{}.create", svc.name)));
            }
            let mut columns: Vec<(String, serde_json::Value)> = svc
                .fields
                .iter()
                .filter(|f| !svc.is_write_excluded_field(f, has_sm))
                .filter_map(|f| inputs.get(&f.name).map(|v| (f.name.clone(), v.clone())))
                .collect();
            // Server-set created_at (literal sentinel the executor resolves to now()).
            columns.push(("created_at".into(), serde_json::Value::Null));
            // Server-set Status = initial state under an SM.
            if let Some(initial) = svc.state_machine.as_ref().map(|m| /* initial accessor */) {
                // confirm the Status column name from the FieldMeaning::Status field;
                // push (status_column_name, json!(initial)).
            }
            Ok(CrudPlan::Create { table, columns, tenant_column: None })
        }
        CrudVerb::Update => {
            if !svc.updatable {
                return Err(crate::Error::VerbNotEnabled(format!("{}.update", svc.name)));
            }
            let id_value = inputs.get("id").cloned().unwrap_or(serde_json::Value::Null);
            let patch = svc
                .fields
                .iter()
                .filter(|f| !svc.is_write_excluded_field(f, has_sm))
                .filter_map(|f| inputs.get(&f.name).map(|v| (f.name.clone(), v.clone())))
                .collect();
            Ok(CrudPlan::Update {
                table,
                id_column: "id".into(),
                id_value,
                patch,
                soft_delete_column: svc.resolved_soft_delete_column().to_string(),
                tenant_column: None,
            })
        }
        CrudVerb::Delete => {
            if !svc.deletable {
                return Err(crate::Error::VerbNotEnabled(format!("{}.delete", svc.name)));
            }
            Ok(CrudPlan::Delete {
                table,
                id_column: "id".into(),
                id_value: inputs.get("id").cloned().unwrap_or(serde_json::Value::Null),
                soft_delete_column: svc.resolved_soft_delete_column().to_string(),
                tenant_column: None,
            })
        }
    }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| agent inputs (`serde_json::Value`) → CrudPlan | untrusted agent payload crosses into the derived plan; column selection happens here |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-241-01 | Tampering | `derive_crud_plan` column selection | mitigate | Column set is filtered by `is_write_excluded_field` (excludes Identifier, CreatedAt, UpdatedAt, **Sensitive**, list, and SM Status). Agent cannot smuggle a Sensitive or server-injected field into the INSERT/UPDATE column set — the gate is the single source of truth shared with the schema builders. Asserted by `derive_crud_plan_create` (Sensitive/server fields absent from `columns`). |
| T-241-02 | Information Disclosure | Sensitive field in derived columns | mitigate | Same gate (Gate C in `is_write_excluded_field`) excludes `FieldMeaning::Sensitive`; no Sensitive value ever enters a CrudPlan, so it cannot reach the audit log downstream. |
| T-241-03 | Spoofing | agent supplies `tenant_id` in `inputs` | mitigate (boundary) | `is_server_injected_field` excludes the tenant column from the writable set, so `derive_crud_plan` never reads `inputs["tenant_id"]`. The `tenant_column` slot is `None` here; actual server-side injection is **Phase 242** (out of scope, by design — D-09). |
| T-241-04 | Tampering | soft-delete addressability | mitigate (derivation half) | `CrudPlan::Update` and `Delete` always carry `soft_delete_column` (unconditional, from `resolved_soft_delete_column()`); the executor (Plan 02) emits `AND <col> IS NULL`. CRUD-03 enforcement is completed in Plan 02; this plan guarantees the predicate column is present in the plan data. |

**Out of scope (DEFERRED to Phase 242):** cross-tenant non-disclosure, `read_write` scope / `.mcp_write_ability` Gate, server-side `tenant_id` injection. SQL-injection parameterization is owned by Plan 02's executor (this plan emits no SQL). The checker should NOT flag these as missing in Plan 01.
</threat_model>
