```rust
pub(crate) fn render_form(el: &Element, spec: &Spec, data: &Value, depth: usize) -> String {
    let props: FormProps = match serde_json::from_value(el.props.clone()) {
        Ok(p) => p,
        Err(e) => {
            return format!(
                "<!-- ferro-json-ui: failed to decode Form props: {} -->",
                html_escape(&e.to_string())
            );
        }
    };

    // Action URL handling (D-15/D-16). FormProps has a required `action: Action` field.
    let (action_url, diagnostic) = match props.action.url.as_deref() {
        Some(u) => (html_escape(u), String::new()),
        None => (
            "#".to_string(),
            format!(
                "<!-- ferro-json-ui: action '{}' has no resolved url -->",
                html_escape(&props.action.handler)
            ),
        ),
    };
    let method = match props.action.method {
        HttpMethod::Get => "GET",
        HttpMethod::Post => "POST",
        HttpMethod::Put => "PUT",
        HttpMethod::Patch => "PATCH",
        HttpMethod::Delete => "DELETE",
    };

    // Fields: el.children (D-05). v1 read props.fields, but FormProps.fields
    // was dropped in Phase 115-02.
    let fields: String = el
        .children
        .iter()
        .map(|cid| render_element(cid, spec, data, depth + 1))
        .collect();

    // Port v1 render_form HTML wrapper. v1 emitted:
    //   <form action="{url}" method="{method}" class="...">{fields}</form>
    // (verify exact class string and any data-* attributes).
    format!(
        "{diagnostic}<form action=\"{action_url}\" method=\"{method}\">{fields}</form>"
    )
}
```

```rust
pub(crate) fn render_switch(el: &Element, _spec: &Spec, data: &Value, _depth: usize) -> String {
    let props: SwitchProps = match serde_json::from_value(el.props.clone()) {
        Ok(p) => p,
        Err(e) => {
            return format!(
                "<!-- ferro-json-ui: failed to decode Switch props: {} -->",
                html_escape(&e.to_string())
            );
        }
    };

    // data_path → checked
    let checked = if let Some(path) = &props.data_path {
        resolve_path(data, path)
            .map(|v| matches!(v, Value::Bool(true)) || matches!(v, Value::Number(n) if n.as_i64() != Some(0)) || matches!(v, Value::String(s) if !s.is_empty()))
            .unwrap_or(false)
    } else {
        props.checked.unwrap_or(false)
    };

    // Render the <input type="checkbox" role="switch" ...> markup per v1.
    let switch_html: String = render_switch_inner(&props, checked);

    // Auto-form wrap if action present (CONTEXT non-obvious behavior).
    if let Some(action) = &el.action {
        let (url, diagnostic) = match action.url.as_deref() {
            Some(u) => (html_escape(u), String::new()),
            None => (
                "#".to_string(),
                format!(
                    "<!-- ferro-json-ui: action '{}' has no resolved url -->",
                    html_escape(&action.handler)
                ),
            ),
        };
        let method = match action.method {
            HttpMethod::Get => "GET",
            HttpMethod::Post => "POST",
            HttpMethod::Put => "PUT",
            HttpMethod::Patch => "PATCH",
            HttpMethod::Delete => "DELETE",
        };
        return format!(
            "{diagnostic}<form action=\"{url}\" method=\"{method}\" data-auto-submit>{switch_html}</form>"
        );
    }
    switch_html
}
```
