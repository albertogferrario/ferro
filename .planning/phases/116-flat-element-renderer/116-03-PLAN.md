pub(crate) fn render_button(el: &Element, _spec: &Spec, _data: &Value, _depth: usize) -> String {
    let props: ButtonProps = match serde_json::from_value(el.props.clone()) {
        Ok(p) => p,
        Err(e) => {
            return format!(
                "<!-- ferro-json-ui: failed to decode Button props: {} -->",
                html_escape(&e.to_string())
            );
        }
    };

    // Render the inner <button> markup verbatim from v1 render_button (lines TBD).
    let inner_html = render_button_inner(&props, el);

    // Wrap in <a> if action exists and method is GET (v1 render_node lines 262-286).
    if let Some(action) = &el.action {
        if matches!(action.method, HttpMethod::Get) {
            let (href, diagnostic) = match &action.url {
                Some(u) => (html_escape(u), String::new()),

                    format!(
                        "<!-- ferro-json-ui: action '{}' has no resolved url -->",
                        html_escape(&action.handler)

    inner_html
