```rust
/// Count column-backed field-builder invocations in projection source.
/// Used to detect under-parsing (D-06): if this exceeds the reconstructed
/// ServiceDef.fields.len(), reconstruction is incomplete.
/// All four builders are column-backed (D-05 vocabulary) — write_only_field
/// MUST be counted (Pitfall 3).
fn count_column_backed_builders(content: &str) -> usize {
    // Strip // line comments so commented-out builder calls are not counted (Pitfall 2).
    let no_comments: String = content
        .lines()
        .map(|line| match line.find("//") {
            Some(pos) => &line[..pos],
            None => line,
        })
        .collect::<Vec<_>>()
        .join("\n");

    let patterns = [
        r"\.field\(",
        r"\.optional_field\(",
        r"\.read_only_field\(",
        r"\.write_only_field\(",
    ];
    patterns
        .iter()
        .map(|p| regex::Regex::new(p).unwrap().find_iter(&no_comments).count())
        .sum()
}
```

Note: `\.field\(` is a substring of the longer builder names? No — the longer names end in `_field(` so `\.field\(` only matches a literal `.field(`, not `.optional_field(`. Verify with the count_all_four test (must equal 4, not more).

Add the three tests to the `#[cfg(test)] mod tests` block.
