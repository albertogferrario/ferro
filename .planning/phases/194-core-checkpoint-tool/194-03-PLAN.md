```rust
/// D-09: fail if any seam fails; else warn if any warns; else pass.
/// not_checked seams are listed in the verdict but NEVER raise status to fail (CHK-03).
fn aggregate_status(seams: &[SeamResult]) -> SeamStatus {
    if seams.iter().any(|s| s.status == SeamStatus::Fail) {
        SeamStatus::Fail
    } else if seams.iter().any(|s| s.status == SeamStatus::Warn) {
        SeamStatus::Warn
    } else {
        SeamStatus::Pass
    }
}

/// D-10: failures before warnings; within a rank, earlier seam first.
/// Dedup by (subject, fix). Format "<fix> (seam: <seam_name>)". Cap at 10.
fn aggregate_next_steps(seams: &[SeamResult]) -> Vec<String> {
    let mut items: Vec<(u8, usize, String, String, String)> = Vec::new();
    for (idx, seam) in seams.iter().enumerate() {
        let rank: u8 = match seam.status {
            SeamStatus::Fail => 0,
            SeamStatus::Warn => 1,
            _ => continue,
        };
        for finding in &seam.findings {
            let entry = format!("{} (seam: {})", finding.fix, seam.seam);
            items.push((rank, idx, finding.subject.clone(), finding.fix.clone(), entry));
        }
    }
    items.sort_by(|a, b| a.0.cmp(&b.0).then(a.1.cmp(&b.1)));
    let mut seen = std::collections::HashSet::new();
    let mut result = Vec::new();
    for (_, _, subject, fix, entry) in items {
        if seen.insert((subject, fix)) {
            result.push(entry);
            if result.len() == 10 {
                break;
            }
        }
    }
    result
}
```

```rust
pub fn execute(project_root: &std::path::Path, name: &str) -> Result<Verdict, String> {

pub(crate) fn run_for(
    project_root: &std::path::Path,
    name: &str,
    now: chrono::DateTime<chrono::Utc>,
) -> Result<Verdict, String> {
    // Path-traversal guard BEFORE any filesystem path is constructed (T-194-01).
    validate_name(name)?;

    // Locate projection (tool-level not-found is an Err; seam-level absence is NotChecked).
    let detail = match super::inspect_projection::execute(project_root, name) {
        super::inspect_projection::InspectResult::Found(d) => d,
        super::inspect_projection::InspectResult::NotFound(nf) => {
            return Err(format!("projection '{}' not found. Available: {:?}", nf.name, nf.available));
        }
    };
    let file_path = project_root.join(&detail.file);
    let content = std::fs::read_to_string(&file_path)
        .map_err(|e| format!("failed to read {}: {e}", detail.file))?;

    // Seam 2 — the only seam producing findings in Phase 194.
    let seam2 = field_to_column_seam(project_root, &detail.service_name, &detail.display_name, &content);

    // Seams 1/3/4/5 — NotChecked stubs (Phase 195 fills these).
    // Seam-cascade rule (STATE.md, documented for Phase 195): seam 1 fail → seams 4+5
    // not_checked (reason seam_1_failed); seam 4 fail → seam 5 not_checked (reason seam_4_failed);
    // seams 2 and 3 run independently. No real execution paths yet — stubs only.
    let stub = |seam: &str| SeamResult {
        seam: seam.to_string(),
        status: SeamStatus::NotChecked,
        source: "checkpoint".to_string(),
        findings: vec![],
        reason: Some("not_implemented_until_phase_195".to_string()),
    };
    let seams = vec![
        stub("validate"),            // seam 1
        seam2,                       // seam 2
        stub("action"),             // seam 3
        stub("render"),             // seam 4
        stub("contracts"),         // seam 5
    ];

    let status = aggregate_status(&seams);
    let next_steps = aggregate_next_steps(&seams);
    let verdict = Verdict { status: status.clone(), projection: name.to_string(), seams, next_steps };

    write_cache(project_root, name, &verdict, now)?;
    Ok(verdict)
}
```

```rust
fn write_cache(
    project_root: &std::path::Path,
    name: &str,
    verdict: &Verdict,
    now: chrono::DateTime<chrono::Utc>,
) -> Result<(), String> {
    #[derive(serde::Serialize)]
    struct CacheEntry<'a> {
        #[serde(flatten)]
        verdict: &'a Verdict,
        ambient_status: &'static str,
        checked_at: chrono::DateTime<chrono::Utc>,
    }
    let ambient_status = if verdict.status == SeamStatus::Pass { "clean" } else { "failing" };
    let entry = CacheEntry { verdict, ambient_status, checked_at: now };

    let cache_dir = project_root.join(".ferro").join("checkpoints");
    std::fs::create_dir_all(&cache_dir).map_err(|e| format!("failed to create cache dir: {e}"))?;
    let path = cache_dir.join(format!("{name}.json"));
    let json = serde_json::to_string_pretty(&entry).map_err(|e| format!("failed to serialize cache: {e}"))?;
    std::fs::write(&path, json).map_err(|e| format!("failed to write cache: {e}"))?;
    Ok(())
}
```
