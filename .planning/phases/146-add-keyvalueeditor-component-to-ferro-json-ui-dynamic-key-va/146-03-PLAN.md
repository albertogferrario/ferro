    ```rust
    //! Runtime module: KeyValueEditor.
    //!
    //! Wires add-row, delete-row, and input-synced hidden-field serialization for
    //! `[data-kv-editor]` wrappers rendered by `render_key_value_editor`.
    //!
    //! - Add-row: clones `[data-kv-row-template].content` and appends to
    //!   `[data-kv-rows]`.
    //! - Delete-row: delegated `click` on `[data-kv-rows]`, matches
    //!   `[data-kv-delete]`, walks up to the nearest `[data-kv-row]` and removes it.
    //! - Input sync: delegated `input` event on `[data-kv-rows]`, reads every
    //!   `[data-kv-row]`'s `[data-kv-key]` and `[data-kv-value]`, serializes as
    //!   a JSON object, writes to the hidden input whose `name` matches
    //!   `data-kv-field`.
    //!
    //! Empty rows (blank key) are excluded from the serialized object.
    //! Duplicate keys take last-write-wins. Vanilla ES5: `var` only, named
    //! function declarations, no arrow functions, no closures capturing `this`.

    pub(super) const SOURCE: &str = r#"
        // ── Key-value editor ──────────────────────────────────────────────────

        function setupKeyValueEditor() {
            var editors = document.querySelectorAll('[data-kv-editor]');
            for (var i = 0; i < editors.length; i++) {
                initKeyValueEditor(editors[i]);
            }
        }

        function initKeyValueEditor(editor) {
            var rowsContainer = editor.querySelector('[data-kv-rows]');
            var addBtn = editor.querySelector('[data-kv-add]');
            var tmpl = editor.querySelector('[data-kv-row-template]');

            if (!rowsContainer || !addBtn || !tmpl) {
                return;
            }

            addBtn.addEventListener('click', function() {
                var clone = tmpl.content.cloneNode(true);
                rowsContainer.appendChild(clone);
                syncHiddenField(editor);
            });

            rowsContainer.addEventListener('click', function(e) {
                var target = e.target;
                // Walk up from target to nearest [data-kv-delete] (if any).
                var delBtn = target.closest ? target.closest('[data-kv-delete]') : null;
                if (!delBtn) {
                    return;
                }
                var row = delBtn.closest('[data-kv-row]');
                if (row && row.parentNode) {
                    row.parentNode.removeChild(row);
                    syncHiddenField(editor);
                }
            });

            rowsContainer.addEventListener('input', function(e) {
                var target = e.target;
                if (!target || !target.hasAttribute) {
                    return;
                }
                if (target.hasAttribute('data-kv-key') || target.hasAttribute('data-kv-value')) {
                    syncHiddenField(editor);
                }
            });
        }

        function syncHiddenField(editor) {
            var fieldName = editor.getAttribute('data-kv-field');
            if (!fieldName) {
                return;
            }
            var hiddenInput = editor.querySelector(
                'input[name="' + fieldName + '"][type="hidden"]'
            );
            if (!hiddenInput) {
                return;
            }
            var rows = editor.querySelectorAll('[data-kv-row]');
            var obj = {};
            for (var i = 0; i < rows.length; i++) {
                var keyEl = rows[i].querySelector('[data-kv-key]');
                var valEl = rows[i].querySelector('[data-kv-value]');
                var k = keyEl ? String(keyEl.value).replace(/^\s+|\s+$/g, '') : '';
                var v = valEl ? valEl.value : '';
                if (k !== '') {
                    // Last-write-wins on duplicate keys (D-10).
                    obj[k] = v;
                }
            }
            hiddenInput.value = JSON.stringify(obj);
        }
    "#;
    ```

    Note on the trim regex: We use `.replace(/^\s+|\s+$/g, '')` instead of
    `.trim()` purely as a defensive ES5 idiom; `String.prototype.trim()` is
    broadly available in any target supporting the rest of ferro-json-ui's
    runtime — either form is acceptable, but this matches the conservative
    idiom used elsewhere in the crate's runtime modules.

    Save the file. Run `cargo fmt --all` (the raw string literal has no Rust

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| DOM → JS | `document.querySelectorAll('[data-kv-editor]')` selects server-rendered wrappers; no untrusted JS source crosses this boundary |
| User input → hidden field value | User-typed key/value flows through `JSON.stringify(obj)` into `hiddenInput.value`; consumed by server-side form parser, not by DOM innerHTML |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-146-JS1 | Tampering | `hiddenInput.value = JSON.stringify(obj)` | mitigate | `JSON.stringify` produces a valid JSON encoding that escapes `"`, `\`, control chars. The server is responsible for re-validating the JSON shape on form submit (same responsibility as every other form field). No `innerHTML` or `eval` appears anywhere in the module |
| T-146-JS2 | Tampering | New row via `tmpl.content.cloneNode(true)` | mitigate | Clones the server-rendered `<template>` — no string concatenation, no user input interpolated into HTML. The cloned node inherits the server's already-escaped default attributes (`value=""`, etc.) |
| T-146-JS3 | Tampering | `editor.querySelector('input[name="' + fieldName + '"][type="hidden"]')` — CSS-selector injection if `fieldName` contains `"]` or similar | accept | `fieldName` is read from `data-kv-field` set by the Rust renderer, which already passed through `html_escape()` in Plan 02. An attacker would need write access to the server to inject there; at that point CSS selector injection is the least of the concerns. Mitigation cost (moving to `getElementsByName` loop with bracket-safe filtering) exceeds the residual risk |
| T-146-JS4 | Denial of service | Runaway add-row click loop | accept | Adding thousands of rows is possible but self-limiting (browser memory); no server-side DoS vector because the hidden field only submits when the form is submitted by the user |
| T-146-JS5 | Information disclosure | Event delegation sees all row-container clicks | accept | Handler only acts on `[data-kv-delete]` and `[data-kv-key]`/`[data-kv-value]` — no data exfiltration path |
</threat_model>
