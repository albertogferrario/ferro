Write pkce.rs (RESEARCH §4):
```rust
use base64::{engine::general_purpose::URL_SAFE_NO_PAD, Engine as _};
use rand::Rng;
use sha2::{Digest, Sha256};
use subtle::ConstantTimeEq;

pub fn generate_auth_code() -> String {
    let bytes: [u8; 32] = rand::thread_rng().gen();
    URL_SAFE_NO_PAD.encode(bytes) // 43 chars, 256 bits
}

pub fn verify_s256(code_verifier: &str, stored_challenge: &str) -> bool {
    let hash = Sha256::digest(code_verifier.as_bytes());
    let recomputed = URL_SAFE_NO_PAD.encode(hash);
    recomputed.as_bytes().ct_eq(stored_challenge.as_bytes()).into()
}
```
`verify_s256` is constant-time (T-11). Note: the S256 challenge is BASE64URL(SHA256(verifier))
per RFC 7636 §4.2 — the `code_challenge_method=plain` rejection (T-01) lives at /authorize
(Plan 04), not here; pkce.rs only computes/compares S256.

```rust
use jsonwebtoken::{decode, DecodingKey, Validation};
pub fn decode_token(token: &str, secret: &[u8], expected_aud: &str)
    -> Result<McpTokenClaims, jsonwebtoken::errors::Error>
{
    let mut validation = Validation::new(Algorithm::HS256);
    validation.algorithms = vec![Algorithm::HS256]; // T-06/T-07 pin — reject alg=none/RS256
    validation.set_audience(&[expected_aud]);        // T-08 audience binding
    validation.validate_exp = true;
    validation.leeway = 0;
    let key = DecodingKey::from_secret(secret);
    Ok(decode::<McpTokenClaims>(token, &key, &validation)?.claims)
}
```
