<objective>
Assemble the full spec JSON Schema (root + `$defs/Element` + `$defs/Action` +
`$defs/Visibility` + `oneOf` discriminated union over all component Props) and
compile the `jsonschema::Validator` ONCE inside `Catalog::build()`. Expose
`catalog.json_schema() -> &serde_json::Value` as a zero-copy accessor.

This plan is the architectural heart of SCHEMA-01. The `oneOf` must use the
idiomatic JSON Schema discriminated-union pattern: each variant pins
`"type": { "const": "X" }` so validators can match by discriminator.

This plan also satisfies SCHEMA-03 by moving validator compilation from the
Plan 02 placeholder into the real full-schema compile path — one call to
`jsonschema::validator_for()` at build time; the `Validator` lives on
`catalog.validator` for the life of the process.

Purpose: `json_schema()` is the public contract external tools (IDEs,
validators, Phase 120 AI structured-output generation) consume. Getting the
oneOf discriminator right here means downstream plans don't have to massage it.

Output: A meta-valid JSON Schema document returned by `catalog.json_schema()`
and a compiled validator on `catalog.validator` ready for Plan 04's validation
pipeline.
</objective>

    ```rust
    /// Hand-assemble the full spec JSON Schema document from per-component schemas.
    ///
    /// Root: `$schema`, `root`, `elements` (HashMap<String, Element>), optional `title` /
    /// `layout` / `data`. `$defs/Element.props` is a `oneOf` over all built-ins and plugins
    /// with a `type` const discriminator per variant (CONTEXT D-13).
    fn assemble_full_schema(
        per_component: &HashMap<String, Value>,
    ) -> Result<Value, CatalogError> {
        let action_schema = sanitize_schema(
            to_value(schema_for!(crate::action::Action))?
        );
        let visibility_schema = sanitize_schema(
            to_value(schema_for!(crate::visibility::Visibility))?
        );

        // Deterministic oneOf — sorted by name (CONTEXT D-18).
        let mut names: Vec<&String> = per_component.keys().collect();
        names.sort();
        let one_of: Vec<Value> = names
            .into_iter()
            .map(|name| {
                let props_schema = per_component[name].clone();
                serde_json::json!({
                    "allOf": [
                        {
                            "type": "object",
                            "required": ["type"],
                            "properties": {
                                "type": { "const": name }
                            }
                        },
                        props_schema
                    ]
                })
            })
            .collect();

        Ok(serde_json::json!({
            "$schema": "https://json-schema.org/draft/2020-12/schema",
            "$id": "ferro-json-ui/v2",
            "type": "object",
            "required": ["$schema", "root", "elements"],
            "properties": {
                "$schema":  { "const": "ferro-json-ui/v2" },
                "root":     { "type": "string", "pattern": "^[A-Za-z_][A-Za-z0-9_-]{0,127}$" },
                "elements": {
                    "type": "object",
                    "additionalProperties": { "$ref": "#/$defs/Element" }
                },
                "title":    { "type": ["string", "null"] },
                "layout":   { "type": ["string", "null"] },
                "data":     true
            },
            "$defs": {
                "Element": {
                    "type": "object",
                    "required": ["type"],
                    "properties": {
                        "type":     { "type": "string" },
                        "props":    { "oneOf": one_of },
                        "children": { "type": "array", "items": { "type": "string" } },
                        "action":   { "$ref": "#/$defs/Action" },
                        "visible":  { "$ref": "#/$defs/Visibility" }
                    }
                },
                "Action": action_schema,
                "Visibility": visibility_schema
            }
        }))
    }
    ```

    ```rust
    fn json_schema_has_spec_envelope_shape() {
        let cat = Catalog::build().expect("build");
        let schema = cat.json_schema();
        assert_eq!(schema["$id"], "ferro-json-ui/v2");
        assert_eq!(schema["type"], "object");
        let required: Vec<&str> = schema["required"]
            .as_array().unwrap().iter().map(|v| v.as_str().unwrap()).collect();
        assert!(required.contains(&"$schema"));
        assert!(required.contains(&"root"));
        assert!(required.contains(&"elements"));
    }

    #[test]
    fn json_schema_has_action_and_visibility_defs() {
        let cat = Catalog::build().expect("build");
        let schema = cat.json_schema();
        assert!(schema["$defs"]["Action"].is_object(), "$defs/Action missing");
        assert!(schema["$defs"]["Visibility"].is_object(), "$defs/Visibility missing");
        assert!(schema["$defs"]["Element"].is_object(), "$defs/Element missing");
    }

    #[test]
    fn json_schema_oneof_covers_all_builtins() {
        let cat = Catalog::build().expect("build");
        let schema = cat.json_schema();
        let one_of = schema["$defs"]["Element"]["properties"]["props"]["oneOf"]
            .as_array()
            .expect("props.oneOf is an array");

        // Extract every const discriminator from the allOf[0] branch.
        let mut discriminators: std::collections::HashSet<String> =
            std::collections::HashSet::new();
        for variant in one_of {
            let c = variant["allOf"][0]["properties"]["type"]["const"]
                .as_str()
                .expect("every variant pins a type const");
            discriminators.insert(c.to_string());
        }

        for name in crate::render::BUILTIN_TYPES.iter() {
            assert!(
                discriminators.contains(*name),
                "oneOf is missing discriminator for '{name}'"
            );
        }

        // Every built-in must be covered; plugins may add extras.
        assert!(
            discriminators.len() >= crate::render::BUILTIN_TYPES.len(),
            "oneOf has fewer variants than BUILTIN_TYPES"
        );
    }

    #[test]
    fn json_schema_is_valid() {
        use jsonschema::draft202012;
        let cat = Catalog::build().expect("build");
        let schema = cat.json_schema();
        assert!(
            draft202012::meta::is_valid(schema),
            "assembled full_schema did not meta-validate as Draft 2020-12"
        );
    }

    #[test]
    fn validator_is_compiled_once_and_usable() {
        let cat = Catalog::build().expect("build");
        // The validator field is private — we prove it's real by validating
        // a minimal valid spec value. If the validator were stale / null /
        // placeholder, this would fail or mis-report.
        let minimal_valid = serde_json::json!({
            "$schema": "ferro-json-ui/v2",
            "root": "r",

        });
        // Should succeed — exact pass-through; no per-element validation yet
        // (that arrives in Plan 04). Full-schema envelope accepts this shape.
        assert!(cat.validator.is_valid(&minimal_valid));
    }

    #[test]
    fn validator_rejects_wrong_schema_version() {
        let cat = Catalog::build().expect("build");
        let wrong_version = serde_json::json!({
            "$schema": "ferro-json-ui/v99-wrong",
            "root": "r",
            "elements": {
                "r": { "type": "Text", "props": { "content": "hi" } }
            }
        });
        assert!(

    }

        assert_eq!(

            serde_json::to_string(cat2.json_schema()).unwrap()
        );
    }
    ```
