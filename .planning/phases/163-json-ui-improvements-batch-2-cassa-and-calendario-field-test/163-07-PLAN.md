```rust
//! v1 → v2 JSON-UI controller migration codemod.
//!
//! Reads a Rust controller file using the v1 `make_node` / `JsonUiView::new`
//! builder pattern, emits a flat JSON spec under `src/views/{module}/{handler}.json`,
//! and rewrites the controller body to call `JsonUi::render_file(...)`.
//!
//! Phase 163 D-09 / D-10 / D-11:
//!   - AST-based (syn 2), not regex (D-11).
//!   - Single file per invocation (D-10).
//!   - Idempotent — re-running on already-migrated source emits a warning + exit 0 (D-11).
//!   - `--dry-run` prints proposed output without writing (D-11).
//!   - Cases the codemod cannot handle produce a TODO marker, not a silent skip (D-09).

use std::path::{Path, PathBuf};

use anyhow::{anyhow, Result};
use serde_json::{json, Map, Value};
use syn::visit::{self, Visit};
use syn::{Expr, ExprCall, File, ImplItemFn, ItemFn};

/// Entry point invoked by the `json-ui:migrate-v1` subcommand.
pub fn run(file: String, dry_run: bool) -> Result<()> {
    let path = PathBuf::from(&file);
    if !path.exists() {
        return Err(anyhow!("controller file not found: {file}"));
    }
    let src = std::fs::read_to_string(&path)?;
    let parsed: File = syn::parse_file(&src)
        .map_err(|e| anyhow!("failed to parse {file} as Rust source: {e}"))?;

    if file_already_migrated(&parsed) {
        eprintln!(
            "warning: {file} appears already migrated (uses JsonUi::render_file). No changes made."
        );
        return Ok(());
    }

    let mut visitor = MigrationVisitor::new(&path);
    visitor.visit_file(&parsed);

    if visitor.specs.is_empty() && visitor.todo_handlers.is_empty() {
        eprintln!(
            "warning: {file} contains no recognizable v1 patterns (make_node / JsonUiView::new). \
             No changes made."
        );
        return Ok(());
    }

    // Compute output spec paths + rewritten controller content.
    let outputs = visitor.finalize(&src)?;

    if dry_run {
        print_dry_run(&outputs);
        return Ok(());
    }

    // Write outputs.
    for SpecOutput { path: spec_path, json } in &outputs.specs {
        if let Some(parent) = spec_path.parent() {
            std::fs::create_dir_all(parent)?;
        }
        std::fs::write(spec_path, json)?;
        eprintln!("wrote {}", spec_path.display());
    }
    std::fs::write(&path, &outputs.rewritten_controller)?;
    eprintln!("rewrote {}", path.display());
    Ok(())
}

/// Detect "already migrated" by checking for any `JsonUi::render_file(...)`
/// call inside an `ItemFn` body.
fn file_already_migrated(parsed: &File) -> bool {
    struct Detector(bool);
    impl<'ast> Visit<'ast> for Detector {
        fn visit_expr_call(&mut self, call: &'ast ExprCall) {
            let s = quote::quote!(#call).to_string();
            if s.contains("JsonUi :: render_file") || s.contains("JsonUi::render_file") {
                self.0 = true;
            }
            visit::visit_expr_call(self, call);
        }
    }
    let mut d = Detector(false);
    d.visit_file(parsed);
    d.0
}

/// Collected output of a migration run.
struct MigrationOutputs {
    specs: Vec<SpecOutput>,
    rewritten_controller: String,
}

struct SpecOutput {
    path: PathBuf,
    json: String,
}

fn print_dry_run(out: &MigrationOutputs) {
    println!("=== dry-run: proposed JSON specs ===");
    for spec in &out.specs {
        println!("--- {} ---", spec.path.display());
        println!("{}", spec.json);
    }
    println!("=== dry-run: rewritten controller ===");
    println!("{}", out.rewritten_controller);
}

/// AST visitor that scans top-level handler `ItemFn`s and impl-method `ImplItemFn`s
/// for `JsonUiView::new` / `make_node` patterns.
struct MigrationVisitor {
    /// Per-handler outcome: either a spec entry or a TODO marker.
    specs: Vec<HandlerMigration>,
    /// Handlers that couldn't be migrated; rewritten file gets a TODO comment.
    todo_handlers: Vec<String>,
    /// Source path; used to compute the `src/views/{module}` prefix.
    controller_path: PathBuf,
}

#[derive(Debug)]
struct HandlerMigration {
    handler_fn_name: String,
    spec_path: PathBuf,
    spec_json: Value,
}

impl MigrationVisitor {
    fn new(controller_path: &Path) -> Self {
        Self {
            specs: Vec::new(),
            todo_handlers: Vec::new(),
            controller_path: controller_path.to_path_buf(),
        }
    }

    fn finalize(self, original_src: &str) -> Result<MigrationOutputs> {
        // For Plan 07 scope: emit ALL recognized specs to disk, rewrite the
        // controller body for those handlers, leave un-migratable handlers
        // intact but prefixed with the TODO comment.
        let mut rewritten = String::from(original_src);
        for handler in &self.todo_handlers {
            let marker = format!("// TODO: ferro json-ui:migrate-v1 could not auto-translate this handler\n");
            rewritten = inject_todo_above_handler(&rewritten, handler, &marker);
        }
        for migration in &self.specs {
            rewritten = rewrite_handler_body(
                &rewritten,
                &migration.handler_fn_name,
                migration.spec_path.to_string_lossy().as_ref(),
            )?;
        }

        let specs: Vec<SpecOutput> = self
            .specs
            .into_iter()
            .map(|m| SpecOutput {
                path: m.spec_path,
                json: serde_json::to_string_pretty(&m.spec_json).unwrap(),
            })
            .collect();
        Ok(MigrationOutputs {
            specs,
            rewritten_controller: rewritten,
        })
    }
}

impl<'ast> Visit<'ast> for MigrationVisitor {
    fn visit_item_fn(&mut self, item: &'ast ItemFn) {
        let handler_name = item.sig.ident.to_string();
        match try_migrate_handler(item, &self.controller_path) {
            HandlerResult::Migrated(spec_path, spec_json) => {
                self.specs.push(HandlerMigration {
                    handler_fn_name: handler_name,
                    spec_path,
                    spec_json,
                });
            }
            HandlerResult::Unsupported => {
                self.todo_handlers.push(handler_name);
            }
            HandlerResult::NotAHandler => {} // skip
        }
        visit::visit_item_fn(self, item);
    }

    fn visit_impl_item_fn(&mut self, item: &'ast ImplItemFn) {
        // Mirror visit_item_fn for methods inside `impl` blocks.
        let handler_name = item.sig.ident.to_string();
        // Wrap in a one-off ItemFn-shaped check using item.block / item.sig.
        // For Plan 07 scope, we only handle free functions. impl methods get
        // the TODO marker.
        if contains_jsonuiview_new(&item.block) {
            self.todo_handlers.push(handler_name);
        }
        visit::visit_impl_item_fn(self, item);
    }
}

#[derive(Debug)]
enum HandlerResult {
    Migrated(PathBuf, Value),
    Unsupported,
    NotAHandler,
}

fn try_migrate_handler(item: &ItemFn, controller_path: &Path) -> HandlerResult {
    // Step 1: find a `JsonUiView::new(title_lit, vec![...])` call in the body.
    let body_tokens = quote::quote!(#item).to_string();
    if !body_tokens.contains("JsonUiView") {
        return HandlerResult::NotAHandler;
    }
    // Step 2: check for known un-translatable patterns.
    if body_tokens.contains("Spec :: builder")
        || body_tokens.contains("Spec::builder")
        || body_tokens.contains("if cond")
        || has_runtime_branch(&item.block)
        || has_dynamic_key(&body_tokens)
    {
        return HandlerResult::Unsupported;
    }
    // Step 3: extract JsonUiView::new args + walk the make_node calls.
    let Some(view) = extract_view_call(&item.block) else {
        return HandlerResult::Unsupported;
    };
    // Step 4: build the flat element map.
    let mut elements = Map::<String, Value>::new();
    let mut root_id: Option<String> = None;
    let nodes_count = view.nodes.len();
    for (idx, node) in view.nodes.into_iter().enumerate() {
        if root_id.is_none() && idx == 0 {
            root_id = Some(node.id.clone());
        }
        let mut el_obj = Map::new();
        el_obj.insert("type".to_string(), Value::String(node.component_type));
        if !node.props.is_empty() {
            el_obj.insert("props".to_string(), Value::Object(node.props));
        }
        if !node.children.is_empty() {
            el_obj.insert(
                "children".to_string(),
                Value::Array(node.children.into_iter().map(Value::String).collect()),
            );
        }
        if let Some(action) = node.action {
            el_obj.insert("action".to_string(), action);
        }
        elements.insert(node.id, Value::Object(el_obj));
    }
    let Some(root) = root_id else {
        return HandlerResult::Unsupported;
    };
    let _ = nodes_count; // suppress unused if not needed

    let mut spec = Map::new();
    spec.insert(
        "$schema".to_string(),
        Value::String("ferro-json-ui/v2".to_string()),
    );
    spec.insert("root".to_string(), Value::String(root));
    if let Some(title) = view.title {
        spec.insert("title".to_string(), Value::String(title));
    }
    if let Some(layout) = view.layout {
        spec.insert("layout".to_string(), Value::String(layout));
    }
    spec.insert("elements".to_string(), Value::Object(elements));

    // Step 5: compute output spec path: src/views/{module}/{handler}.json
    // where {module} is derived from the controller's path stem (e.g.,
    // src/controllers/auth.rs -> auth/{handler}.json).
    let module_name = controller_path
        .file_stem()
        .and_then(|s| s.to_str())
        .unwrap_or("unknown")
        .to_string();
    let handler_name = item.sig.ident.to_string();
    let spec_path: PathBuf = ["src", "views", &module_name, &format!("{handler_name}.json")]
        .iter()
        .collect();

    HandlerResult::Migrated(spec_path, Value::Object(spec))
}

#[derive(Debug)]
struct ExtractedView {
    title: Option<String>,
    layout: Option<String>,
    nodes: Vec<ExtractedNode>,
}

#[derive(Debug)]
struct ExtractedNode {
    id: String,
    component_type: String,
    props: Map<String, Value>,
    children: Vec<String>,
    action: Option<Value>,
}

fn extract_view_call(_block: &syn::Block) -> Option<ExtractedView> {
    // Implementation note for Wave 4 executor:
    //
    // Walk the block looking for `JsonUiView::new(title, vec![...]).layout("...")`.
    // - title is the first arg's String literal.
    // - layout is the chained `.layout(...)` arg's String literal (Option).
    // - The vec![...] body is a list of `make_node(id_lit, Component::X(struct_literal))`
    //   or `make_node_with_action(id_lit, Component::X(struct_literal), Action::post(name_lit))`.
    //
    // For each node:
    //   - id = the literal `id_lit`
    //   - component_type = the Component::X variant name
    //   - props = struct_literal fields converted to JSON values (only literals
    //     and arrays of literals; expressions or function calls → Unsupported).
    //   - children = if the props contain `fields: vec![nested make_node...]`
    //     or `buttons: vec![...]`, recurse and collect the inner IDs into
    //     `children`, and push each inner node into the same flat list.
    //   - action = if the call is `make_node_with_action(..., Action::post("handler"))`,
    //     emit `{"handler": "handler", "method": "POST"}`. Mirror for Action::get/etc.
    //
    // Use syn::visit::Visit to walk the block. Build the recursive collector by
    // returning Vec<ExtractedNode> from a helper.
    //
    // For Plan 07 scope, the executor implements this helper to handle the
    // RESEARCH §"Codemod transformation rule" auth.rs fixture exactly.
    None
}

fn has_runtime_branch(_block: &syn::Block) -> bool {
    // Returns true if any `if`-let/`match` returns different ComponentNode/Component
    // values. Simple heuristic for Plan 07: if the block contains any `if cond`
    // or `match` *inside* the JsonUiView::new vec![...], mark Unsupported.
    false
}

fn has_dynamic_key(body_tokens: &str) -> bool {
    // Look for format!("{}_{}", ...) patterns inside make_node first arg.
    body_tokens.contains("format!")
        && (body_tokens.contains("make_node !") || body_tokens.contains("make_node ("))
}

fn contains_jsonuiview_new(block: &syn::Block) -> bool {
    quote::quote!(#block).to_string().contains("JsonUiView")
}

// ---- Source rewriting helpers ----

fn inject_todo_above_handler(src: &str, handler_name: &str, marker: &str) -> String {
    // Find "fn <handler_name>(" at the start of a line and insert `marker` above it.
    let needle = format!("fn {handler_name}(");
    let mut out = String::with_capacity(src.len() + marker.len());
    let mut last = 0usize;
    if let Some(pos) = src.find(&needle) {
        // Walk back to the start of line.
        let line_start = src[..pos].rfind('\n').map(|n| n + 1).unwrap_or(0);
        out.push_str(&src[last..line_start]);
        out.push_str(marker);
        out.push_str(&src[line_start..]);
        last = src.len();
    }
    if last == 0 {
        return src.to_string();
    }
    out
}

fn rewrite_handler_body(src: &str, handler_name: &str, spec_path: &str) -> Result<String> {
    // Replace the body of `fn <handler_name>(...)` with a single statement:
    // `JsonUi::render_file("<spec_path>", json!({}))`.
    //
    // Strategy: find `fn <handler_name>(`, then walk braces to find matching
    // `}`, then replace the inner body. Use a hand-rolled brace counter (not
    // regex). If we cannot find the function, return the original source
    // unchanged with a warning.
    let needle = format!("fn {handler_name}(");
    let Some(start) = src.find(&needle) else {
        eprintln!("warning: handler {handler_name} not found at rewrite time");
        return Ok(src.to_string());
    };
    // Find the opening `{` after the signature.
    let Some(brace_off) = src[start..].find('{') else {
        return Ok(src.to_string());
    };
    let body_start = start + brace_off;
    let mut depth = 0;
    let mut body_end = body_start;
    for (i, ch) in src[body_start..].char_indices() {
        match ch {
            '{' => depth += 1,
            '}' => {
                depth -= 1;
                if depth == 0 {
                    body_end = body_start + i + 1;
                    break;
                }
            }
            _ => {}
        }
    }
    if body_end == body_start {
        return Ok(src.to_string());
    }
    let new_body = format!(
        "{{\n    JsonUi::render_file(\"{spec_path}\", serde_json::json!({{}}))\n}}"
    );
    let mut out = String::with_capacity(src.len());
    out.push_str(&src[..body_start]);
    out.push_str(&new_body);
    out.push_str(&src[body_end..]);
    Ok(out)
}
```

   - `title`: the string literal from `args[0]` if it is `Expr::Lit(ExprLit { lit: Lit::Str(_) })`. Otherwise return `None` (Unsupported).
   - `vec_expr`: the `vec![...]` macro invocation in `args[1]`. Parse its tokens via `syn::parse2::<syn::ExprArray>` or by walking the macro's tokens. For Plan 07 scope, use the standard pattern: `expr_to_make_node_list(&args[1]) -> Option<Vec<ExtractedNode>>` which iterates the elements.

```rust
fn parse_make_node(expr: &Expr) -> Option<ExtractedNode> {
    let Expr::Call(call) = expr else { return None };
    let fn_name = path_ident_str(&call.func)?;
    let with_action = match fn_name.as_str() {
        "make_node" => false,
        "make_node_with_action" => true,
        _ => return None,
    };
    let args = &call.args;
    let id = lit_str(&args[0])?;
    let component_expr = &args[1];
    let (component_type, props_map, nested_children) = parse_component_expr(component_expr)?;
    let action = if with_action {
        let action_expr = &args[2];
        Some(parse_action_expr(action_expr)?)
    } else {
        None
    };
    Some(ExtractedNode {
        id,
        component_type,
        props: props_map,
        children: nested_children,
        action,
    })
}
```

5. `parse_component_expr(&Expr) -> Option<(String, Map<String, Value>, Vec<String>)>` matches `Component::Variant(StructLit { fields })`:
   - `Component::PageHeader(PageHeaderProps { title: "Login".to_string(), ..Default::default() })` → `("PageHeader", {"title": "Login"}, [])`.
   - If a field is `fields: vec![make_node(...)]` (or `buttons: vec![...]`, `children: vec![...]`), recurse: each inner node's ID is appended to the `children` Vec<String>, and the inner node is collected into the OUTER node list (return the IDs only here; the caller stores the flat list).
6. `parse_action_expr(&Expr) -> Option<Value>` matches `Action::post("name")`, `Action::get("name")`, etc.:
   - Emits `{"handler": "name", "method": "POST"}` JSON object.
7. `lit_str(&Expr) -> Option<String>` extracts a string literal value, handling both `"str".to_string()` (a MethodCall) and plain `"str"` (an ExprLit).

Use the `syn` types: `Expr`, `ExprCall`, `ExprStruct`, `ExprMacro`, `ExprMethodCall`, `Lit`, `LitStr`, `FieldValue`. For struct literal field iteration: `expr_struct.fields.iter()`.

```rust
/// Best-effort cwd guard: restores cwd on drop. Tests must serialize cwd changes
/// — annotate with `#[serial_test::serial]` if a `serial_test` dep is already in
/// the workspace; otherwise rely on `cargo test --test json_ui_migrate_v1 -- --test-threads=1`.
struct ChangeCwd {
    original: PathBuf,
}

impl ChangeCwd {
    fn new(new_dir: &std::path::Path) -> Self {
        let original = std::env::current_dir().unwrap();
        std::env::set_current_dir(new_dir).unwrap();
        Self { original }
    }
}

impl Drop for ChangeCwd {
    fn drop(&mut self) {
        let _ = std::env::set_current_dir(&self.original);
    }
}
```
