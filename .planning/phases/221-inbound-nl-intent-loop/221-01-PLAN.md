    ```rust
    use crate::renderer::{render_exposed_tools, McpContext};
    use ferro_projections::ServiceDef;

    pub fn render_tool_descriptions(
        services: &[ServiceDef],
        ctx: &McpContext,
    ) -> Result<String, ferro_projections::Error> {
        let tools = render_exposed_tools(services, ctx)?;
        let lines: Vec<String> = tools.iter().map(|t| {
            let props: Vec<&str> = t.input_schema
                .get("properties").and_then(|v| v.as_object())
                .map(|m| m.keys().map(|k| k.as_str()).collect())
                .unwrap_or_default();
            format!("- {}: {} [args: {}]",
                t.name,
                t.description.as_deref().unwrap_or(""),
                props.join(", "))
        }).collect();
        Ok(lines.join("\n"))
    }
    ```
